// @vitest-environment node
import { describe, it, expect, vi, beforeEach } from 'vitest'
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js'

const generateCarouselMock = vi.fn()
const deductCreditAdminMock = vi.fn()
const getCreditBalanceAdminMock = vi.fn()
const getUserTierMock = vi.fn()
const checkProjectLimitMock = vi.fn()
const insertMock = vi.fn(() => Promise.resolve({ error: null }))
const fromMock = vi.fn(() => ({ insert: insertMock }))

vi.mock('@/lib/services/carousel', () => ({ generateCarousel: generateCarouselMock }))
vi.mock('@/lib/credits/check-credits', () => ({
  deductCreditAdmin: deductCreditAdminMock,
  getCreditBalanceAdmin: getCreditBalanceAdminMock,
  getUserTier: getUserTierMock,
  checkProjectLimit: checkProjectLimitMock,
  FREE_TIER_PROJECT_LIMIT: 3,
}))
vi.mock('@/lib/supabase/admin', () => ({ createAdminClient: () => ({ from: fromMock }) }))

import { registerTools } from '@/lib/mcp/tools'

function getCreateCarousel() {
  const server = new McpServer({ name: 'test', version: '0.0.0' })
  registerTools(server)
  return ((server as unknown) as { _registeredTools: Record<string, { callback: (args: unknown, extra: unknown) => Promise<{ isError?: boolean; content: { text: string }[] }> }> })._registeredTools.create_carousel
}

const VALID_AUTH = { authInfo: { extra: { userId: 'user-1', keyId: 'key-1' } } }

describe('create_carousel tool (MCP-TOOL-02)', () => {
  beforeEach(() => {
    vi.clearAllMocks()
    insertMock.mockClear()
    fromMock.mockClear()
    getUserTierMock.mockResolvedValue('creator')
    checkProjectLimitMock.mockResolvedValue(true)
    getCreditBalanceAdminMock.mockResolvedValue(10)
    deductCreditAdminMock.mockResolvedValue(true)
  })

  it('happy path: deducts 1 credit, calls generateCarousel, returns { projectId, slideCount, dashboardUrl }', async () => {
    generateCarouselMock.mockResolvedValue({ projectId: 'proj-123', slideCount: 7 })
    const tool = getCreateCarousel()
    const res = await tool.callback({ topic: 'productivity', templateId: 'highlight-bar' }, VALID_AUTH)
    expect(res.isError).toBeFalsy()
    const body = JSON.parse(res.content[0].text)
    expect(body.projectId).toBe('proj-123')
    expect(body.slideCount).toBe(7)
    expect(body.dashboardUrl).toMatch(/\/project\/proj-123$/)
    expect(deductCreditAdminMock).toHaveBeenCalledWith(expect.anything(), 'user-1', 'carousel_generation', expect.stringContaining('productivity'), undefined, 1)
    expect(generateCarouselMock).toHaveBeenCalledTimes(1)
  })

  it('insufficient credits: returns INSUFFICIENT_CREDITS in-band error (NOT thrown)', async () => {
    getCreditBalanceAdminMock.mockResolvedValue(0)
    const tool = getCreateCarousel()
    const res = await tool.callback({ topic: 'x', templateId: 'highlight-bar' }, VALID_AUTH)
    expect(res.isError).toBe(true)
    expect(JSON.parse(res.content[0].text).error).toBe('INSUFFICIENT_CREDITS')
    expect(deductCreditAdminMock).not.toHaveBeenCalled()
    expect(generateCarouselMock).not.toHaveBeenCalled()
  })

  it('refund on failure: when generateCarousel throws, inserts a refund credit_transactions row', async () => {
    generateCarouselMock.mockRejectedValue(new Error('Gemini timeout'))
    const tool = getCreateCarousel()
    const res = await tool.callback({ topic: 'x', templateId: 'highlight-bar' }, VALID_AUTH)
    expect(res.isError).toBe(true)
    expect(JSON.parse(res.content[0].text).error).toBe('GENERATION_FAILED')
    expect(fromMock).toHaveBeenCalledWith('credit_transactions')
    expect(insertMock).toHaveBeenCalledWith(expect.objectContaining({ user_id: 'user-1', type: 'refund', amount: 1 }))
  })
})

          hook_text: 'new', body_text: 'new', not_allowed: 'rejected',
          style_overrides: { hook_text_color: '#fff' },

      },
      VALID_AUTH
    )

    expect(slideData[0].body_text).toBe('new')

    const tool = getEditSlide()

    expect(JSON.parse(res.content[0].text).error).toBe('VALIDATION_ERROR')
    expect(updateProjectServiceMock).not.toHaveBeenCalled()

export function registerTools(server: McpServer): void {
  // ─── Tool 1: list_templates (free) ─────────────────────────────────────────
  server.registerTool(
    'list_templates',
    {
      description: [
        'List the available carousel visual styles (templates).',
        'Call this FIRST to discover valid templateId values BEFORE using create_carousel.',
        'Returns id, name, thumbnail, textPosition, and textAlign for each template.',
        'No credit cost. No prerequisites beyond the standard Bearer-token MCP auth.',
        'Typical workflow: list_templates → create_carousel(topic, templateId) → edit_slide(...) → render_carousel(projectId).',
      ].join(' '),
      inputSchema: {},
    },
    async () => {
      const data = templates.map(({ id, name, thumbnail, textPosition, textAlign }) => ({
        id, name, thumbnail, textPosition, textAlign,
      }))
      return successContent({ templates: data })
    }
  )

  // ─── Tool 2: create_carousel (1 credit) ───────────────────────────────────
  server.registerTool(
    'create_carousel',
    {
      description: [
        'Create a new carousel from a topic. Generates AI text content + background images and saves a project.',
        'Costs 1 credit (same as POST /api/v1/carousel). Returns { projectId, slideCount, dashboardUrl } so the user can open the result in browser.',
        'PREREQUISITE: call list_templates first and pass a valid templateId (no default — bold-minimal was removed in v2.5).',
        'Optional slideCount (3-10, default 7; free tier capped at 3). Optional imageSource: stock (Pexels, fast) or ai (Imagen, slower, no extra credit).',
        'On insufficient credits, returns INSUFFICIENT_CREDITS in-band error (not an exception).',
      ].join(' '),
      inputSchema: {
        topic: z.string().min(1).max(500).describe('Carousel topic. Example: "5 productivity tips for remote workers".'),
        templateId: z.string().min(1).describe('Template ID from list_templates. REQUIRED. Example: "highlight-bar".'),
        slideCount: z.number().int().min(3).max(10).optional().describe('Slide count 3-10. Default 7. Free plan capped at 3.'),
        imageSource: z.enum(['stock', 'ai']).optional().describe('"stock" = Pexels (free, fast). "ai" = Imagen (slower, higher visual quality). Default "stock".'),
      },
    },
    async (args, extra) => {
      const userId = getUserId(extra)
      if (!userId) return errorContent('UNAUTHORIZED', 'Missing auth context')

      const { topic, templateId, slideCount, imageSource } = args as {
        topic: string; templateId: string; slideCount?: number; imageSource?: 'stock' | 'ai'
      }
      const supabase = createAdminClient()
      const validSlideCount = slideCount ?? 7
      const validImageSource = imageSource ?? 'stock'

      // Free-tier project-count cap (mirrors /api/v1/carousel/route.ts:64-71)
      const tier = await getUserTier(userId)
      const projectAllowed = await checkProjectLimit(userId, tier)
      if (!projectAllowed) {
        return errorContent('PROJECT_LIMIT_EXCEEDED', `Free plan is limited to ${FREE_TIER_PROJECT_LIMIT} projects. Upgrade for unlimited.`)
      }

      // Free-tier slide-count cap (mirrors /api/v1/carousel/route.ts:54-61)
      if (tier === 'free' && validSlideCount > 3) {
        return errorContent('VALIDATION_ERROR', 'Free plan is limited to 3 slides per carousel. Upgrade to hobbyist or creator for up to 10.')
      }

      // Credit balance check (mirrors /api/v1/carousel/route.ts:73-76)
      const balance = await getCreditBalanceAdmin(supabase, userId)
      if (balance < 1) {
        return errorContent('INSUFFICIENT_CREDITS', 'You need at least 1 credit to create a carousel.', { creditsRemaining: balance })
      }

      // Atomic deduct (mirrors /api/v1/carousel/route.ts:78-89)
      const deducted = await deductCreditAdmin(
        supabase, userId, 'carousel_generation', `Carousel: ${topic}`, undefined, 1
      )
      if (!deducted) {
        return errorContent('INSUFFICIENT_CREDITS', 'Credit deduction failed (race condition or insufficient balance).')
      }

      try {
        const result = await generateCarousel(supabase, userId, {
          topic,
          templateId,
          slideCount: validSlideCount,
          imageSource: validImageSource,
        })
        return successContent({
          projectId: result.projectId,
          slideCount: result.slideCount,
          dashboardUrl: `${getAppUrl()}/project/${result.projectId}`,
        })
      } catch (err) {
        // Refund — mirrors /api/v1/carousel/route.ts:106-114
        await supabase.from('credit_transactions').insert({
          user_id: userId,
          type: 'refund',
          amount: 1,
          description: 'Carousel generation failed (MCP)',
        })
        return errorContent('GENERATION_FAILED', err instanceof Error ? err.message : 'Unknown error')
      }
    }
  )

  // ─── Tool 3: edit_slide (free) ─────────────────────────────────────────────
  server.registerTool(
    'edit_slide',
    {
      description: [
        'Edit a single slide in an existing carousel project. No credit cost.',
        'Provide projectId from create_carousel, the zero-based slideIndex, and a patch object containing the fields to change.',
        'Patch supports: hook_text, body_text, hook_segments, body_segments, image_query, background_url (must be https://),',
        'style_overrides (deep-merged with existing), custom_texts, custom_images. Other fields are silently rejected.',
        'IMPORTANT: image_query updates the metadata field only — it does NOT auto-fetch a new image.',
        'To swap an image, set background_url directly to an https:// URL.',
        'After editing, call render_carousel to get fresh download URLs.',
      ].join(' '),
      inputSchema: {
        projectId: z.string().uuid().describe('Project ID returned from create_carousel.'),
        slideIndex: z.number().int().min(0).describe('Zero-based slide index.'),
        patch: z.record(z.string(), z.unknown()).describe('Object containing one or more of the 9 allowed fields. Other keys are ignored.'),
      },
    },
    async (args, extra) => {
      const userId = getUserId(extra)
      if (!userId) return errorContent('UNAUTHORIZED', 'Missing auth context')

      const { projectId, slideIndex, patch } = args as {
        projectId: string; slideIndex: number; patch: Record<string, unknown>
      }

      // Filter to whitelisted fields (mirrors PATCH route line 73-78)
      const patchFields: Record<string, unknown> = {}
      for (const field of ALLOWED_PATCH_FIELDS) {
        if (field in patch) patchFields[field] = patch[field]
      }
      if (Object.keys(patchFields).length === 0) {
        return errorContent('VALIDATION_ERROR', `At least one of ${ALLOWED_PATCH_FIELDS.join(', ')} must be provided.`)
      }

      // Validate background_url scheme (mirrors PATCH route line 84-87, T-41-04)
      if (typeof patchFields.background_url === 'string' && !patchFields.background_url.startsWith('https://')) {
        return errorContent('VALIDATION_ERROR', 'background_url must use https://')
      }

      const supabase = createAdminClient()
      let project
      try {
        project = await getProject(supabase, userId, projectId)
      } catch {
        return errorContent('NOT_FOUND', 'Project not found')
      }

      const slideData = (project.slide_data as Record<string, unknown>[] | null) ?? []
      if (slideIndex >= slideData.length) {
        return errorContent('NOT_FOUND', 'Slide not found')
      }
      const existingSlide = slideData[slideIndex]

      // Deep-merge style_overrides (mirrors PATCH route line 105-117)
      let mergedStyleOverrides = existingSlide.style_overrides
      if (patchFields.style_overrides && typeof patchFields.style_overrides === 'object') {
        mergedStyleOverrides = {
          ...(existingSlide.style_overrides as Record<string, unknown> | undefined),
          ...(patchFields.style_overrides as Record<string, unknown>),
        }
      }
      const updatedSlide = {
        ...existingSlide,
        ...patchFields,
        ...(patchFields.style_overrides ? { style_overrides: mergedStyleOverrides } : {}),
      }

      const updatedArray = [...slideData]
      updatedArray[slideIndex] = updatedSlide
      await updateProjectService(supabase, userId, projectId, { slide_data: updatedArray })
      return successContent({ slide: updatedSlide })
    }
  )

  // ─── Tool 4: render_carousel (free) ────────────────────────────────────────
  server.registerTool(
    'render_carousel',
    {
      description: [
        'Get fresh download URLs for every slide in a carousel project, plus a single ZIP URL containing all slides.',
        'Use this AFTER create_carousel or edit_slide. No credit cost.',
        'Returns { slides: [{ index, downloadUrl }], zipUrl } — URLs only, NEVER inline binary image data.',
        'URLs point at the same host as the MCP server and re-authenticate on download via the user project ownership.',
        'Open dashboardUrl from create_carousel for an interactive preview, or fetch downloadUrl/zipUrl for the raw PNGs.',
      ].join(' '),
      inputSchema: {
        projectId: z.string().uuid().describe('Project ID returned from create_carousel.'),
      },
    },
    async (args, extra) => {
      const userId = getUserId(extra)
      if (!userId) return errorContent('UNAUTHORIZED', 'Missing auth context')

      const { projectId } = args as { projectId: string }
      const supabase = createAdminClient()
      let project
      try {
        project = await getProject(supabase, userId, projectId)
      } catch {
        return errorContent('NOT_FOUND', 'Project not found')
      }
      const slides = (project.slide_data as unknown[] | null) ?? []
      if (slides.length === 0) {
        return errorContent('NOT_FOUND', 'Project has no slides')
      }

      const baseUrl = getAppUrl()
      const slideUrls = slides.map((_, index) => ({
        index,
        downloadUrl: `${baseUrl}/api/render?projectId=${projectId}&slide=${index}&download`,
      }))
      return successContent({
        slides: slideUrls,
        zipUrl: `${baseUrl}/api/v1/projects/${projectId}/download`,
      })
    }
  )
}
