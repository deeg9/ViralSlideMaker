# Plan 49-02: Reducer Actions + Editor Integration + Dual-Write Save

**Phase:** 49 — Schema Unification & Migration
**Plan:** 02 of 02
**Goal:** Wire the adapter from Plan 49-01 into the editor reducer, slide loading, and save path. Add 4 new reducer actions for text-box manipulation. Selection state stored outside history. Dual-write keeps legacy fields populated.

**Depends on:** Plan 49-01 (TextBox type + adapter must exist)

## Decisions

- **Selection state location:** Keep `EditorHistory.present` as `Slide[]` (no shape change). Store `selectedTextBoxId: string | null` as a separate React state in the editor hook (`useSlideEditor` if it exists, otherwise `slide-editor.tsx` directly). Selection never undoes — matches Figma/Canva.
- **Eager migration on load:** When `useSlideEditor` initializes (or when `SET_SLIDES` fires with slides that lack `text_boxes`), run `legacyToTextBoxes()` for each slide and stash on `slide.text_boxes`. UI reads from `text_boxes` only.
- **Dual-write on save:** When the user clicks Save, run `textBoxesToLegacy()` for each slide and merge the legacy fields back into the slide before persisting. Both shapes ship to Supabase.
- **Action creators:** Following the `crypto.randomUUID()` pattern from existing custom_texts/images actions.

### T-02 — Reducer handlers

**File:** `src/lib/editor/editor-reducer.ts`
**Action:** Edit

Add 3 new cases to the reducer switch:

```typescript
case 'ADD_TEXT_BOX': {
  const slides = state.present.map((slide, i) => {
    if (i !== action.slideIndex) return slide
    const newBox: TextBox = {
      id: crypto.randomUUID(),
      role: action.role,
      text: action.defaults?.text ?? 'New Text',
      position: action.defaults?.position ?? defaultPositionForRole(action.role),
      ...action.defaults,
    }
    return {
      ...slide,
      text_boxes: [...(slide.text_boxes ?? []), newBox],
    }
  })
  return pushHistory(state, slides)
}

case 'UPDATE_TEXT_BOX': {
  const slides = state.present.map((slide, i) => {
    if (i !== action.slideIndex) return slide
    const text_boxes = (slide.text_boxes ?? []).map((tb) =>
      tb.id === action.textBoxId ? { ...tb, ...action.updates } : tb
    )
    return { ...slide, text_boxes }
  })
  return pushHistory(state, slides)
}

case 'REMOVE_TEXT_BOX': {
  const slides = state.present.map((slide, i) => {
    if (i !== action.slideIndex) return slide
    const text_boxes = (slide.text_boxes ?? []).filter(
      (tb) => tb.id !== action.textBoxId
    )
    return { ...slide, text_boxes }
  })
  return pushHistory(state, slides)
}
```

Add a helper `defaultPositionForRole(role: TextBoxRole): TextBoxPosition`:
- 'hook' → `{ anchor: 'top' }`
- 'body' → `{ anchor: 'bottom' }`
- 'custom' → `{ anchor: 'custom', x: 20, y: 50, width: 60 }` (mirrors existing custom_text defaults at editor-reducer.ts:88-95)

Imports: `TextBox`, `TextBoxRole`, `TextBoxPosition` from `@/lib/types/slide`.

**Acceptance:** `npx tsc --noEmit` clean.

test('full lifecycle: legacy slide → load → ADD_TEXT_BOX → UNDO → REDO → save → reload renders identical to legacy', () => {
  const originalLegacy: Slide = { /* shape 4 fixture from Plan 49-01 */ }
  const slidesWithBoxes = [originalLegacy].map((s) => ({
    ...s,
    text_boxes: legacyToTextBoxes(s),
  }))

  let state: EditorHistory = { past: [], present: slidesWithBoxes, future: [] }

  state = editorReducer(state, { type: 'ADD_TEXT_BOX', slideIndex: 0, role: 'custom', defaults: { text: 'Bonus' } })
  expect(state.present[0].text_boxes!.length).toBe(originalLegacy.custom_texts ? originalLegacy.custom_texts.length + 3 : 3)

  state = editorReducer(state, { type: 'UNDO' })
  expect(state.present[0].text_boxes!.length).toBe(slidesWithBoxes[0].text_boxes!.length)

  state = editorReducer(state, { type: 'REDO' })
  expect(state.present[0].text_boxes![state.present[0].text_boxes!.length - 1].text).toBe('Bonus')

  // Save dual-write
  const saved = state.present.map(saveDualWrite)  // helper that mirrors the production save path
  expect(saved[0].hook_text).toBe(originalLegacy.hook_text)
  expect(saved[0].text_boxes).toBeDefined()
})
