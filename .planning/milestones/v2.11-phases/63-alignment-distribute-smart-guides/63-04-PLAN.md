    ```typescript
    /**
     * Phase 63 (SNAP-04 + DIFF-03): equal-spacing "==" detector for the smart-
     * guide overlay. When N>=3 elements have CONSECUTIVE equal gaps along an
     * axis (within epsilon), Figma renders a pink double-equals indicator
     * between each pair. This pure helper computes the match.
     *
     * Pure module — no React, no DOM, no I/O.
     */
    export interface EqualSpacingMatch {
      gapSize: number
      segments: Array<{ from: number; to: number }>  // gap rect positions in axis units
      elementIds: string[]                            // sorted by pos, length N
    }

    export function detectEqualSpacing(
      elements: Array<{ id: string; pos: number; size: number }>,
      epsilon = 1,
    ): EqualSpacingMatch | null {
      if (elements.length < 3) return null
      const sorted = [...elements].sort((a, b) => a.pos - b.pos)
      const gaps: number[] = []
      for (let i = 0; i < sorted.length - 1; i++) {
        const gap = sorted[i + 1].pos - (sorted[i].pos + sorted[i].size)
        if (gap < 0) return null  // overlap → no spacing match
        gaps.push(gap)
      }
      const first = gaps[0]
      for (const g of gaps) {
        if (Math.abs(g - first) > epsilon) return null
      }
      const segments = gaps.map((_, i) => ({
        from: sorted[i].pos + sorted[i].size,
        to: sorted[i + 1].pos,
      }))
      return {
        gapSize: first,
        segments,
        elementIds: sorted.map(e => e.id),
      }
    }
    ```

       ```typescript
       function runSnapForDrag(draggingId: string, x: number, y: number, width: number, height: number): { x: number; y: number } {
         if (modifierState.get().cmd) {
           guideOverlayRef.current?.clear()
           return { x, y }
         }
         const idx = snapIndexRef.current
         if (!idx) return { x, y }

         // Compute dragging entity edges (REF px) — for multi-element drag, x/y/width/height are the union AABB
         const dLeft = x, dRight = x + width, dCenterX = x + width / 2
         const dTop = y, dBottom = y + height, dCenterY = y + height / 2

         // X axis: pick best snap among the 3 dragging X-edges
         const xCandidates = [dLeft, dCenterX, dRight]
         let bestX: HysteresisResult = { snapped: false, snappedTo: null, delta: 0, nextWasSnapped: null }
         let bestXAbs = Infinity
         for (const cand of xCandidates) {
           const r = snapWithHysteresis(cand, idx.xEdges, wasSnappedXRef.current)
           if (r.snapped && Math.abs(r.delta) < bestXAbs) { bestXAbs = Math.abs(r.delta); bestX = r }
         }
         wasSnappedXRef.current = bestX.nextWasSnapped
         const snappedX = bestX.snapped ? x + bestX.delta : x

         // Y axis: same shape
         // ... bestY computation ...
         wasSnappedYRef.current = bestY.nextWasSnapped
         const snappedY = bestY.snapped ? y + bestY.delta : y

         // Build guide-line list for overlay
         const xLines: GuideLine[] = bestX.snapped && bestX.snappedTo !== null ? [{ pos: bestX.snappedTo }] : []
         const yLines: GuideLine[] = bestY.snapped && bestY.snappedTo !== null ? [{ pos: bestY.snappedTo }] : []

         // Equal-spacing detection: when N>=3 elements (excluding dragging) align along the drag's axis
         // Build candidate elements for axis-equal check:
         const slideElements: Array<{id:string; xPos:number; xSize:number; yPos:number; ySize:number}> = ... (walk slide minus dragging ids)
         const xMatch = detectEqualSpacing(slideElements.map(e => ({ id: e.id, pos: e.xPos, size: e.xSize })))

         return { x: snappedX, y: snappedY }
       }
       ```
