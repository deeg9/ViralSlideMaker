Group drag detection logic (insert before the existing per-element callback in onDragStop):
```typescript
function shouldGroupDrag(slide, selection, currentlyInsideGroup, draggedElementId): string[] | null {
  // Returns the group member ids to BULK_TRANSLATE, or null if this is a single-element drag.
  if (currentlyInsideGroup !== null) return null  // inside group → leaf drag, not group drag
  if (selection.ids.length < 2) return null
  const draggedEl = /* find by id across text_boxes ∪ custom_images ∪ watermark */
  if (!draggedEl?.groupId) return null
  // Are ALL selected ids members of the same group?
  const memberIds = getGroupMembers(slide, draggedEl.groupId)
  const memberSet = new Set(memberIds)
  const selectionSet = new Set(selection.ids)
  if (memberSet.size !== selectionSet.size) return null
  for (const id of selectionSet) {
    if (!memberSet.has(id)) return null
  }
  return memberIds
}
```

### `$mod+Shift+g` (Cmd+Shift+G) handler

```typescript
'$mod+Shift+g': (e) => {
  if (isEditingTarget(e.target)) return
  if (!currentSlide) return
  if (selection.ids.length === 0) return
  // Find unique groupIds among selected elements.
  const allElements = [
    ...(currentSlide.text_boxes ?? []),
    ...(currentSlide.custom_images ?? []),
    ...(currentSlide.watermark ? [currentSlide.watermark] : []),
  ]
  const selectedSet = new Set(selection.ids)
  const groupIds = new Set<string>()
  for (const el of allElements) {
    if (selectedSet.has(el.id) && el.groupId) groupIds.add(el.groupId)
  }
  if (groupIds.size === 0) return                    // no grouped elements selected → no-op
  // Union of all members across all selected groups.
  const idsToUngroup: string[] = []
  for (const gid of groupIds) {
    idsToUngroup.push(...getGroupMembers(currentSlide, gid))
  }
  e.preventDefault()
  dispatch({
    type: 'UNGROUP',
    slideIndex: selectedIndex,
    ids: idsToUngroup,
  })
},
```

### Group drag onDragStop dispatcher

The existing per-element onDragStop callbacks in slide-preview.tsx (4 sites: hook, body, custom_text, custom_image) currently fire `onPositionChange` / `onCustomTextPositionChange` / `onCustomImagePositionChange`. Phase 62-03 wraps them with a group-aware branch:

```tsx
onDragStop={(e, data) => {
  // Existing: compute newX/newY in canvas %, derive deltaX/deltaY from
  // the per-element drag-start ref (Plan 61-02 pattern).

  // Phase 62 GROUP-09: if this drag started a group drag, dispatch
  // BULK_TRANSLATE in ONE call. Otherwise fall through to the existing
  // per-element callback.
  const groupIds = shouldGroupDrag(slide, selection, currentlyInsideGroup, ci.id)
  if (groupIds) {
    onGroupDrag?.({
      slideIndex,        // currently passed via index prop
      ids: groupIds,
      deltaX: newX - startX,  // % of canvas width
      deltaY: newY - startY,  // % of canvas height
    })
    /* clear drag-start ref + return — DO NOT also fire the per-element callback */
    return
  }

  // Existing per-element callback path:
  onCustomImagePositionChange?.(ci.id, { x: newX, y: newY })
}}
```

The new prop on SlidePreview: `onGroupDrag?: (payload: { slideIndex: number; ids: string[]; deltaX: number; deltaY: number }) => void`. Threaded from `slide-editor.tsx` which dispatches BULK_TRANSLATE.
