```tsx
onMouseDown={(e) => {
  e.stopPropagation()
  const target = /* resolve element by role/id at this wrapper */
  if (!target) return

  // SEL-02 / SEL-03 — Shift-click toggle (existing, leave unchanged)
  if (e.shiftKey && onSelectionChange) { ... return }

  // GROUP-10 (Figma model): when target has a groupId AND the user is NOT
  // inside the group, select ALL group members. When inside the group, fall
  // through to SEL-01 single-click leaf select.
  if (
    target.groupId &&
    currentlyInsideGroup !== target.groupId &&  // not entered into the same group
    onSelectionChange
  ) {
    const memberIds = getGroupMembers(slide, target.groupId)
    if (memberIds.length >= 2) {
      onSelectionChange({ ids: memberIds, primary: target.id })
      // legacy single-id slot dual-write keeps right rail consistent
      onTextBoxSelect?.(null)         // group selection isn't single-text-box
      onCustomTextSelect?.(null)
      onCustomImageSelect?.(null)
      return
    }
    // memberIds.length === 1 (orphan groupId) — fall through to single-select
  }

  // SEL-01 — single-click replaces (existing, leave unchanged)
  onSelectionChange?.({ ids: [target.id], primary: target.id })
  /* legacy single-id setter */
}}
onDoubleClick={(e) => {
  // GROUP-07: enter the group on double-click of any group member.
  e.stopPropagation()
  if (target.groupId) {
    onEnterGroup?.(target.id, slide)
  }
}}
```
