    ```typescript
    // Phase 54 (D-09, D-10, D-11, D-12, SLOP-01/02/04): Single source of truth
    // for AI-slop guardrails. Imported by every Imagen call site, the Pexels
    // search wrapper, and the premium-tier system-prompt assembly.

    export const IMAGEN_ADDENDUM: string =
      "Do not generate gradient backgrounds. " +
      "Do not generate glossy 3D objects or abstract color blobs. " +
      "Do not generate purple-to-blue, sunset orange-to-pink, or teal-to-navy gradients. " +
      "Generate documentary-style photographs with naturalistic lighting."

    export const AI_HOOK_VARIETY_ADDENDUM: string =
      "Do not start hooks with 'In a world where', 'Did you know', or 'Imagine if'. " +
      "Vary hook openings across slides in the same carousel — no two slides should start with the same word or phrase pattern."

    export const PEXELS_POSITIVE_TERMS = ' natural lighting, candid, editorial, portrait' as const

    export const PEXELS_NEGATIVE_ALT_TERMS = [
      'meeting', 'office', 'team', 'smiling', 'diverse group',
      'business meeting', 'group of people',
    ] as const

    export function pexelsQueryWithGuardrails(query: string): string {
      return `${query}${PEXELS_POSITIVE_TERMS}`
    }

    export function filterPexelsResultsByAlt<T extends { alt?: string | null }>(
      photos: T[],
    ): T[] {
      const filtered = photos.filter((p) => {
        const alt = (p.alt ?? '').toLowerCase()
        return !PEXELS_NEGATIVE_ALT_TERMS.some((neg) => alt.includes(neg))
      })
      // Per D-11: never block generation — if filtering empties the result, return original.
      return filtered.length > 0 ? filtered : photos
    }
    ```

    ```sql
    -- Phase 54 (D-19, METRIC-02): admin/analytics view for ZIP download rate.
    -- Not user-facing. Surfaced in admin dashboard. Computes
    --   rate = downloads_in_window / projects_created_in_window
    -- Defaults to last 30 days when called without explicit window.

    CREATE OR REPLACE FUNCTION public.zip_download_rate(
      window_start timestamptz DEFAULT (now() - interval '30 days'),
      window_end   timestamptz DEFAULT now()
    )
    RETURNS TABLE (
      window_start    timestamptz,
      window_end      timestamptz,
      downloads       bigint,
      projects_created bigint,
      rate            numeric
    )
    LANGUAGE sql
    SECURITY DEFINER
    SET search_path = ''
    AS $$
      SELECT
        zip_download_rate.window_start,
        zip_download_rate.window_end,
        COALESCE(d.cnt, 0) AS downloads,
        COALESCE(p.cnt, 0) AS projects_created,
        CASE WHEN COALESCE(p.cnt, 0) = 0
          THEN 0::numeric
          ELSE ROUND(COALESCE(d.cnt, 0)::numeric / p.cnt::numeric, 4)
        END AS rate
      FROM (
        SELECT count(*) AS cnt FROM public.carousel_events
        WHERE event_type = 'download_zip'
          AND occurred_at >= zip_download_rate.window_start
          AND occurred_at <  zip_download_rate.window_end
      ) d
      CROSS JOIN (
        SELECT count(*) AS cnt FROM public.projects
        WHERE created_at >= zip_download_rate.window_start
          AND created_at <  zip_download_rate.window_end
      ) p;
    $$;

    GRANT EXECUTE ON FUNCTION public.zip_download_rate(timestamptz, timestamptz) TO authenticated;

    -- View wraps the function with default window for admin dashboard convenience:
    CREATE OR REPLACE VIEW public.v_zip_download_rate AS
      SELECT * FROM public.zip_download_rate();

    GRANT SELECT ON public.v_zip_download_rate TO authenticated;
    ```

    ```typescript
    import { NextRequest, NextResponse } from 'next/server'
    import { createAdminClient } from '@/lib/supabase/admin'
    import { createClient } from '@/lib/supabase/server'

    // Phase 54 (METRIC-01, D-17, D-18): track download_zip events. Body schema:
    //   { slide_count: number, template_id: string | null, user_tier: string }
    // Inserts into carousel_events with event_data payload. Authenticated only.

    export async function POST(
      req: NextRequest,
      { params }: { params: Promise<{ id: string }> },
    ): Promise<NextResponse> {
      const { id: projectId } = await params

      // 1. Authenticate
      const supabase = await createClient()
      const { data: { user } } = await supabase.auth.getUser()
      if (!user) return NextResponse.json({ error: 'unauthenticated' }, { status: 401 })

      // 2. Verify project ownership
      const { data: project } = await supabase
        .from('projects')
        .select('id, user_id, template_id')
        .eq('id', projectId)
        .maybeSingle()
      if (!project || project.user_id !== user.id) {
        return NextResponse.json({ error: 'not_found' }, { status: 404 })
      }

      // 3. Parse body
      let body: { slide_count?: number; template_id?: string | null; user_tier?: string }
      try {
        body = await req.json()
      } catch {
        return NextResponse.json({ error: 'invalid_body' }, { status: 400 })
      }
      const slideCount = typeof body.slide_count === 'number' ? body.slide_count : 0
      const templateId = typeof body.template_id === 'string' ? body.template_id : (project.template_id ?? null)
      const userTier = typeof body.user_tier === 'string' ? body.user_tier : 'free'

      // 4. Insert event (admin client bypasses RLS for the audit-style insert)
      const admin = createAdminClient()
      const { error } = await admin.from('carousel_events').insert({
        event_type: 'download_zip',
        project_id: projectId,
        published_carousel_id: null,
        event_data: {
          project_id: projectId,
          slide_count: slideCount,
          template_id: templateId,
          user_tier: userTier,
        },
      })
      if (error) {
        return NextResponse.json({ error: 'insert_failed', detail: error.message }, { status: 500 })
      }

      return NextResponse.json({ tracked: true }, { status: 200 })
    }
    ```

    ```typescript
    import JSZip from 'jszip'
    import { saveAs } from 'file-saver'

    export async function downloadSlidesAsZip(
      projectId: string,
      slideCount: number,
      projectTitle: string,
    ): Promise<void> {
      const zip = new JSZip()

      // Fetch all slides in parallel
      const fetchPromises = Array.from({ length: slideCount }, (_, i) =>
        fetch(`/api/render?projectId=${projectId}&slide=${i}`)
          .then((res) => {
            if (!res.ok) throw new Error(`Failed to fetch slide ${i + 1}`)
            return res.blob()
          }),
      )

      const blobs = await Promise.all(fetchPromises)

      blobs.forEach((blob, i) => {
        const num = String(i + 1).padStart(2, '0')
        zip.file(`slide-${num}.png`, blob)
      })

      const content = await zip.generateAsync({ type: 'blob' })

      // Phase 54 (METRIC-01): fire-and-forget download tracking. NEVER block
      // the user's download on tracking failure. Server reads template_id and
      // user_tier from the project record server-side; client provides
      // slide_count for cheap denormalization.
      try {
        void fetch(`/api/v1/projects/${projectId}/track-download`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ slide_count: slideCount }),
          keepalive: true,
        })
      } catch {
        // swallow — tracking failure must NOT prevent the user's download
      }

      saveAs(content, `${projectTitle}.zip`)
    }
    ```
