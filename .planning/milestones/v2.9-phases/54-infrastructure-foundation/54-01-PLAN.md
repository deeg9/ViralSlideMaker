```typescript
export type TreatmentPreset =
  | { kind: 'bw'; contrast?: number }                                 // monochrome film B&W
  | { kind: 'duotone-cream'; intensity?: number }                     // cream-tinted duotone
  | { kind: 'duotone-cobalt'; intensity?: number }                    // cobalt-tinted duotone
  | { kind: 'contrast-bump'; gamma?: number; linear?: [number, number] } // gentle contrast pop

export type Palette = {
  dominant: { r: number; g: number; b: number; hex: string }
  cluster1: { r: number; g: number; b: number; hex: string }   // from sharp.stats().channels (D-02)
  cluster2: { r: number; g: number; b: number; hex: string }
  cluster3: { r: number; g: number; b: number; hex: string }
}

export type TreatmentResult = { url: string; palette: Palette }

// Idempotent — calling twice with same (sourceUrl, spec) returns cached URL without re-running sharp.
// Cache key: SHA-256 hex of `${sourceUrl}|${JSON.stringify(spec)}` (D-03).
// Storage path: `${userId}/treated/${sha256}.png` in `slide-backgrounds-treated` bucket.
export async function treatPhoto(
  sourceUrl: string,
  spec: TreatmentPreset,
  userId: string
): Promise<TreatmentResult>
```
