Ship the **deterministic asset library** for Phase 56 templates: 5 grain PNGs (generated by `scripts/build-textures.ts`) + 14 decorative overlay assets (12 SVG + 2 PNG) in `public/overlays/`. This plan ships ONLY the static assets and the generator script — no schema changes, no template files, no resolver helpers. Phase 56 plans 02 and 03 consume these paths.

**Why this is Wave 1:** Templates and resolver helpers in plans 02 / 03 reference these paths in their config. Committing the assets first means subsequent plans can reference real files (not 404s) at write-time and CI doesn't temporarily break.

**Determinism is load-bearing.** `npm run textures` invoked twice MUST produce byte-identical PNGs. Without this, every CI run would diff the grain PNGs and produce noise (spec brief constraint, CONTEXT D-18). Achieved via seeded noise: every randomized step uses a precomputed buffer derived from `crypto.createHash('sha256').update(seedString).digest()`. NO `Math.random()`.

**SVG authoring approach** (CONTEXT D-16): hand-written SVG path data with locked fill colors and viewBox dimensions per UI-SPEC. Each SVG is small (<2 KB), readable, editable. Drop shadows use SVG `<filter feDropShadow>` (Satori-compatible — SVG filters work; CSS `filter:` properties do not).

// Same seed string → byte-identical output across runs.

import { createHash } from 'crypto'
import {
  buildGrainSubtle,
  buildGrainMedium,
  buildGrainHeavy,
  buildPaperTexture,
  buildXeroxGrain,
} from './build-textures'

const sha256 = (buf: Buffer): string => createHash('sha256').update(buf).digest('hex')

describe('build-textures determinism (TPL-05, D-18)', () => {
  it('buildGrainSubtle is byte-identical across two invocations', async () => {
    const a = await buildGrainSubtle()
    const b = await buildGrainSubtle()

  })

  it('buildGrainMedium is byte-identical across two invocations', async () => {
    const a = await buildGrainMedium()
    const b = await buildGrainMedium()
    expect(sha256(a)).toBe(sha256(b))
  })

  it('buildGrainHeavy is byte-identical across two invocations', async () => {
    const a = await buildGrainHeavy()
    const b = await buildGrainHeavy()
    expect(sha256(a)).toBe(sha256(b))
  })

  it('buildPaperTexture is byte-identical across two invocations', async () => {
    const a = await buildPaperTexture()
    const b = await buildPaperTexture()

interface BayerOpts { seed: string; spacing: number; matrixSize: number; foregroundRgba: { r: number, g: number, b: number, a: number } }

async function buildBayerHalftone(opts: BayerOpts): Promise<Buffer> {
  // Allocate 1080*1920*4 bytes (RGBA), default all to transparent (alpha=0).
  // For each grid cell at (col*spacing, row*spacing), draw a small filled circle
  // of radius ~spacing*0.4. Within each circle, modulate alpha by a Bayer matrix
  // pattern (deterministic — no random).
  // The Bayer matrix is hardcoded:
  //   matrixSize=4: [[0,8,2,10],[12,4,14,6],[3,11,1,9],[15,7,13,5]] / 16
  //   matrixSize=8: standard 8x8 Bayer (64 unique values 0..63 / 64)
  // This produces a clean halftone-dot pattern, byte-identical across runs
  // because every value comes from the matrix lookup, not Math.random().
  // ...implementation per the spec...
  // Wrap the raw RGBA buffer with sharp and emit PNG.
  const rawBuf = Buffer.alloc(CANVAS.W * CANVAS.H * 4, 0)  // all transparent
  // ...iterate grid + write pixels via the Bayer matrix...
  return await sharp(rawBuf, { raw: { width: CANVAS.W, height: CANVAS.H, channels: 4 } })
    .png()
    .toBuffer()
}
