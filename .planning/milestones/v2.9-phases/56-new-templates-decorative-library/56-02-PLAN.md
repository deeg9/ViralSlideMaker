### renderDecorativeOverlay resolver export (D-14)
```typescript
// src/lib/render/resolve-overlay.ts — exported helper consumed by both renderers.
// Resolves to /overlays/{asset}.svg for SVG assets and /overlays/{asset}.png for halftone.
// PNG list: ['halftone-dots-light', 'halftone-dots-dense'] (the 2 PNG assets — others are SVG).

export function renderDecorativeOverlay(
  asset: DecorativeAsset,
  position: { x: number; y: number },           // pixel coords on 1080×1920 canvas
  options?: { rotation?: number; scale?: number; opacity?: number }
): React.ReactNode {
  const PNG_ASSETS: ReadonlyArray<DecorativeAsset> = ['halftone-dots-light', 'halftone-dots-dense']
  const ext = PNG_ASSETS.includes(asset) ? 'png' : 'svg'
  const src = `/overlays/${asset}.${ext}`
  const transform = options?.rotation || options?.scale
    ? `rotate(${options.rotation ?? 0}deg) scale(${options.scale ?? 1})`
    : undefined
  return React.createElement('img', {
    'data-layer': 'decorative',
    'data-asset': asset,
    src,
    style: {
      position: 'absolute',
      top: position.y,
      left: position.x,
      transform,
      opacity: options?.opacity ?? 1,
      pointerEvents: 'none',
    },
  })
}
```
