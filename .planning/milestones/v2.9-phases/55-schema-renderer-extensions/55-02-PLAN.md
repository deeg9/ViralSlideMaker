Add a lint-friendly Jest test that scans `resolve-overlay.ts` for forbidden CSS properties (`filter:`, `backdrop-filter:`, `mix-blend-mode:`) — Pitfall #2 protection (Satori silently drops these).

**Critical invariant:** Existing 8 templates declare `grainMode` and `vignette` as `undefined`. The helpers MUST be conditionally invoked (`template.grainMode ? renderGrain(template.grainMode) : null`) so no template gets a new layer in Phase 55. This keeps the byte-identical guarantee from Plan 55-01 intact at the rendered-output level.

**Why no PNG asset commits in this plan (D-10 reaffirmed):** Phase 56 ships `scripts/build-textures.ts` which generates the 5 grain PNGs. Pre-committing stub PNGs creates merge churn. Existing 8 templates do NOT declare `grainMode` → `<img>` with `/textures/grain-*.png` is never injected into the DOM at runtime in Phase 55 → the 404s are not user-visible. Resolver helper unit tests validate the JSX shape without actually loading the asset.

**Why renderer wiring lands in Phase 55, not Phase 56:** TPLREND-05 mandates that BOTH renderers consume any new layer from the SAME resolver module. Adding `renderGrain` to `resolve-overlay.ts` without consumer wiring would let Phase 56's first-template work introduce subtle drift between Satori vs React preview. By landing the wiring NOW with conditional invocation, we lock the DOM-order contract before any consumer template exists — the Phase 56 templates can't pick a different insertion point even by accident.

```typescript
/**
 * Phase 55 (D-09, D-10, D-11, D-13, PARITY-02): grain/noise overlay layer.
 * Returns an <img> element pointing to a placeholder path that resolves at
 * render time. Phase 56 ships the actual PNG assets via scripts/build-textures.ts;
 * Phase 55 only ships the helper. Existing 8 templates leave template.grainMode
 * undefined → this helper is never invoked at runtime in Phase 55.
 *
 * Opacity map (D-11) — tunable in Phase 56 once real PNGs exist:
 *   subtle: 0.04   medium: 0.08   heavy: 0.14   paper: 0.18   xerox: 0.22
 *
 * Element shape: absolutely positioned full-canvas <img> (1080×1920).
 * No <picture>, no srcSet, no responsive sizing — Satori needs deterministic
 * pixel-perfect <img> markup.
 */
export function renderGrain(
  intensity: 'subtle' | 'medium' | 'heavy' | 'paper' | 'xerox'
): React.ReactNode

/**
 * Phase 55 (D-09, D-12, D-13, PARITY-02): radial vignette layer.
 * Returns a <div> with inline backgroundImage: radial-gradient(...).
 * Pure CSS, no filter/backdrop-filter/mix-blend-mode (Pitfall #2: Satori
 * silently drops these). Strength 0-1 maps to outer rgba alpha.
 *
 * Element shape: absolutely positioned full-canvas <div> (1080×1920).
 * Inner gradient stop at 40% transparent; outer at 100% rgba(0,0,0,strength).
 */
export function renderVignette(strength: number): React.ReactNode
```

```typescript
// In resolve-overlay.ts (after pickContrastColor function, before bottomCardBackground):

const GRAIN_OPACITY_MAP = {
  subtle: 0.04,
  medium: 0.08,
  heavy: 0.14,
  paper: 0.18,
  xerox: 0.22,
} as const

export function renderGrain(
  intensity: 'subtle' | 'medium' | 'heavy' | 'paper' | 'xerox',
): React.ReactNode {
  // Tile a static noise PNG at low opacity. Phase 56 ships these via
  // scripts/build-textures.ts → /public/textures/grain-${intensity}.png.
  // In Phase 55 the helper is never invoked at runtime (no template declares
  // grainMode), so 404 risk is moot.
  return React.createElement('img', {
    'data-layer': 'grain',
    src: `/textures/grain-${intensity}.png`,
    style: {
      position: 'absolute',
      top: 0,
      left: 0,
      width: CANVAS.W,
      height: CANVAS.H,
      objectFit: 'cover',
      opacity: GRAIN_OPACITY_MAP[intensity],
      pointerEvents: 'none',
    },
  })
}

export function renderVignette(strength: number): React.ReactNode {
  // Pure radial-gradient — verified Satori-supported. NO CSS filter,
  // NO backdrop-filter, NO mix-blend-mode. Pitfall #2 + D-24 ban.
  return React.createElement('div', {
    'data-layer': 'vignette',
    style: {
      position: 'absolute',
      top: 0,
      left: 0,
      width: CANVAS.W,
      height: CANVAS.H,
      backgroundImage: `radial-gradient(ellipse at center, transparent 40%, rgba(0,0,0,${strength}) 100%)`,
      pointerEvents: 'none',
    },
  })
}
```

// src/lib/render/__tests__/no-css-filter.test.ts
import { describe, it, expect } from 'vitest'
import { readFileSync } from 'node:fs'
import { resolve } from 'node:path'

describe('Phase 55 (D-24, D-25, PARITY-04): no CSS filter properties in resolver', () => {
  const resolverPath = resolve(__dirname, '..', 'resolve-overlay.ts')
  const source = readFileSync(resolverPath, 'utf8')

  it('does NOT contain "filter:" property (Pitfall #2 — Satori silently drops)', () => {
    // Allow `// filter:` in COMMENTS that document the ban; ban only the property usage.
    // Strip comments first, then assert no `filter:` token appears.
    const stripped = source
      .replace(/\/\/.*$/gm, '')          // single-line comments
      .replace(/\/\*[\s\S]*?\*\//g, '')  // block comments
    expect(stripped).not.toMatch(/(?<!backdrop-)filter\s*:/)
  })

  it('does NOT contain "backdrop-filter:" property', () => {
    const stripped = source
      .replace(/\/\/.*$/gm, '')
      .replace(/\/\*[\s\S]*?\*\//g, '')
    expect(stripped).not.toMatch(/backdrop-filter\s*:/)
  })

  it('does NOT contain "mix-blend-mode:" property', () => {
    const stripped = source
      .replace(/\/\/.*$/gm, '')
      .replace(/\/\*[\s\S]*?\*\//g, '')
    expect(stripped).not.toMatch(/mix-blend-mode\s*:/)
  })
})

describe('Phase 55: renderGrain', () => {
  it('returns an <img> with the correct src for each intensity', () => {
    const intensities = ['subtle', 'medium', 'heavy', 'paper', 'xerox'] as const
    for (const intensity of intensities) {
      const el = renderGrain(intensity) as React.ReactElement
      expect(el.type).toBe('img')
      expect(el.props.src).toBe(`/textures/grain-${intensity}.png`)
      expect(el.props.style.position).toBe('absolute')
      expect(el.props.style.width).toBe(1080)
      expect(el.props.style.height).toBe(1920)
      expect(el.props.style.pointerEvents).toBe('none')
    }
  })

  it('opacity map matches D-11 locked values', () => {
    const expected = { subtle: 0.04, medium: 0.08, heavy: 0.14, paper: 0.18, xerox: 0.22 }
    for (const [intensity, opacity] of Object.entries(expected)) {
      const el = renderGrain(intensity as keyof typeof expected) as React.ReactElement
      expect(el.props.style.opacity).toBe(opacity)
    }
  })
})

describe('Phase 55: renderVignette', () => {
  it('returns a <div> with radial-gradient backgroundImage', () => {
    const el = renderVignette(0.5) as React.ReactElement
    expect(el.type).toBe('div')
    expect(el.props.style.backgroundImage).toMatch(/^radial-gradient\(.*ellipse at center.*transparent 40%.*rgba\(0,\s*0,\s*0,\s*0\.5\).*100%\)/)
    expect(el.props.style.position).toBe('absolute')
    expect(el.props.style.width).toBe(1080)
    expect(el.props.style.height).toBe(1920)
    expect(el.props.style.pointerEvents).toBe('none')
  })

  it('does NOT use CSS filter property (Pitfall #2 + D-24)', () => {
    const el = renderVignette(0.5) as React.ReactElement
    expect(el.props.style).not.toHaveProperty('filter')
    expect(el.props.style).not.toHaveProperty('backdropFilter')
    expect(el.props.style).not.toHaveProperty('mixBlendMode')
  })

  it('strength is interpolated into rgba alpha', () => {
    const el = renderVignette(0.7) as React.ReactElement
    expect(el.props.style.backgroundImage).toContain('rgba(0,0,0,0.7)')
  })
})
