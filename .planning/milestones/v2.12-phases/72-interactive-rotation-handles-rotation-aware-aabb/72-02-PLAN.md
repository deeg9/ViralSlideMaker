    ```tsx
    const [localRotation, setLocalRotation] = useState<string>(
      typeof decoration.rotation === 'number' ? String(decoration.rotation) : '0'
    )
    // Round-trip: re-sync local state when the prop changes (drag handle commits etc).
    useEffect(() => {
      setLocalRotation(
        typeof decoration.rotation === 'number' ? String(decoration.rotation) : '0'
      )
    }, [decoration.rotation, decoration.id])

    const commitRotation = () => {
      const trimmed = localRotation.trim()
      if (trimmed === '') {
        // Revert to current decoration value
        setLocalRotation(typeof decoration.rotation === 'number' ? String(decoration.rotation) : '0')
        return
      }
      const parsed = Number(trimmed)
      if (!Number.isFinite(parsed)) {
        setLocalRotation(typeof decoration.rotation === 'number' ? String(decoration.rotation) : '0')
        return

      const clamped = Math.max(-180, Math.min(180, parsed))
      if (clamped === (decoration.rotation ?? 0)) return  // no-op if unchanged
      onUpdate({ rotation: clamped })
      setLocalRotation(String(clamped))  // reflect clamp back to UI
    }
    ```
