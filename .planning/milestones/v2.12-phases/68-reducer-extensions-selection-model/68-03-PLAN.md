    import { describe, it, expect } from 'vitest'
    import { readFileSync } from 'fs'
    import { resolve } from 'path'

    const REDUCER_PATH = resolve(__dirname, '..', 'editor-reducer.ts')
    const REDUCER_SRC = readFileSync(REDUCER_PATH, 'utf-8')

    const ALL_4_SCOPES = ['text_boxes', 'custom_images', 'watermark', 'decorations'] as const

    /**
     * Slice the source body of a single switch case. Returns the lines from
     * `case 'X':` (exclusive of the case label) up to the next `case ` or
     * the closing `}` at the same indent level. Tolerates nested braces by
     * counting `{` / `}` from the opening brace of the case body.
     *
     * Strip line-comments (`// ...`) to avoid the "comment mentions scope but
     * code does not" false-green class. Block comments retained for now —
     * the existing v2.11 codebase doesn't put scope literals inside /* */
     * blocks. Test SELF-CTRL-1 pins this contract.
     */
    function sliceCaseBody(src: string, actionType: string): string {
      // Locate the `case 'ACTION_TYPE':` line.
      // Capture from the next char after the `{` until the matching `}`
      // by counting brace depth.
      const caseStart = src.indexOf(`case '${actionType}':`)
      if (caseStart === -1) {
        throw new Error(`[reducer-scope-symmetry] case '${actionType}' not found`)
      }
      // Find first '{' after the case label
      const openIdx = src.indexOf('{', caseStart)
      if (openIdx === -1) {
        throw new Error(`[reducer-scope-symmetry] case '${actionType}' has no opening brace`)
      }
      let depth = 1
      let i = openIdx + 1
      while (i < src.length && depth > 0) {
        const ch = src[i]
        if (ch === '{') depth++
        else if (ch === '}') depth--
        i++
      }
      const body = src.slice(openIdx + 1, i - 1)
      // Strip line comments
      return body.split('\n').map((line) => {
        const idx = line.indexOf('//')
        return idx === -1 ? line : line.slice(0, idx)
      }).join('\n')
    }

    function mentionsScope(body: string, scope: string): boolean {
      return body.includes(scope)
    }

    describe('Phase 68 PARITY-15: reducer scope symmetry audit (PRA1..PRA12)', () => {
      // PRA1..PRA3: single-scope decoration-only actions
      const SINGLE_SCOPE = [
        ['PRA1', 'ADD_DECORATION'],
        ['PRA2', 'UPDATE_DECORATION'],
        ['PRA3', 'REMOVE_DECORATION'],
      ] as const
      for (const [pra, action] of SINGLE_SCOPE) {
        it(`${pra}: ${action} case body mentions 'decorations'`, () => {
          const body = sliceCaseBody(REDUCER_SRC, action)
          expect(mentionsScope(body, 'decorations'), `${action} body must mention 'decorations'`).toBe(true)
        })
      }

      // PRA4..PRA12: multi-scope actions — must mention all 4 scope literals
      const MULTI_SCOPE = [
        ['PRA4',  'BULK_TRANSLATE'],
        ['PRA5',  'BULK_DELETE'],
        ['PRA6',  'DUPLICATE_ELEMENTS_AT'],
        ['PRA7',  'GROUP'],
        ['PRA8',  'UNGROUP'],
        ['PRA9',  'SET_LOCKED'],
        ['PRA10', 'SET_HIDDEN'],
        ['PRA11', 'ALIGN'],
        ['PRA12', 'Z_ORDER'],
      ] as const
      for (const [pra, action] of MULTI_SCOPE) {
        it(`${pra}: ${action} case body mentions all 4 AddressableScope values`, () => {
          const body = sliceCaseBody(REDUCER_SRC, action)
          const missing = ALL_4_SCOPES.filter((s) => !mentionsScope(body, s))
          expect(
            missing,
            `${action} body missing scope(s): ${missing.join(', ')}.\nBody (first 600 chars):\n${body.slice(0, 600)}`,
          ).toEqual([])
        })
      }

      // Self-controls — pin the methodology
      it('SELF-CTRL-1: helper detects missing scope on a synthetic broken body', () => {
        const synthetic = `
          // text_boxes branch — but no other scope literals
          const next = slide.text_boxes ?? []
        `.trim()
        const missing = ALL_4_SCOPES.filter((s) => !mentionsScope(synthetic, s))
        // 3 of 4 must be reported missing (only 'text_boxes' is present)
        expect(missing).toEqual(['custom_images', 'watermark', 'decorations'])
      })

      it('SELF-CTRL-2: helper passes a synthetic body that mentions all 4 scopes', () => {
        const synthetic = `
          slide.text_boxes; slide.custom_images; slide.watermark; slide.decorations
        `.trim()
        const missing = ALL_4_SCOPES.filter((s) => !mentionsScope(synthetic, s))
        expect(missing).toEqual([])
      })

      it('SELF-CTRL-3: line-comment stripper actually strips // comments', () => {
        // A body that mentions 'decorations' ONLY in a // comment is a self-
        // invalidating grep gate (the grep finds the literal but the runtime
        // code never touches the field). The slicer must strip the comment
        // and report 'decorations' as MISSING.
        const synthetic = `
          // decorations branch — TODO not yet implemented
          const next = slide.text_boxes
        `.trim()
        // Re-run the slicer's strip step on this synthetic input
        const stripped = synthetic.split('\n').map((line) => {
          const idx = line.indexOf('//')
          return idx === -1 ? line : line.slice(0, idx)
        }).join('\n')
        // After stripping, only 'text_boxes' remains

        expect(stripped.includes('text_boxes')).toBe(true)
      })
    })
