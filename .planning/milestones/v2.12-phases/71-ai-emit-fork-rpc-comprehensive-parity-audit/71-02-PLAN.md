```sql
CREATE OR REPLACE FUNCTION public._strip_locked_from_slide_data(p_slide_data jsonb)
RETURNS jsonb
LANGUAGE plpgsql IMMUTABLE SET search_path = ''
AS $$
DECLARE
  result jsonb := '[]'::jsonb;
  slide jsonb; slide_out jsonb;
  arr jsonb; arr_out jsonb;
  el jsonb; el_out jsonb;
  arr_field text;
BEGIN
  IF p_slide_data IS NULL OR jsonb_typeof(p_slide_data) <> 'array' THEN
    RETURN p_slide_data;
  END IF;
  FOR slide IN SELECT * FROM jsonb_array_elements(p_slide_data) LOOP
    slide_out := slide;
    -- Strip locked from each element of text_boxes / custom_images / custom_texts.
    -- Phase 71 (MIGR-01) adds 'decorations' to this ARRAY[...] list.
    FOREACH arr_field IN ARRAY ARRAY['text_boxes', 'custom_images', 'custom_texts']
    LOOP
      arr := slide_out -> arr_field;
      IF arr IS NOT NULL AND jsonb_typeof(arr) = 'array' THEN
        arr_out := '[]'::jsonb;
        FOR el IN SELECT * FROM jsonb_array_elements(arr) LOOP
          el_out := el #- '{locked}';
          arr_out := arr_out || jsonb_build_array(el_out);
        END LOOP;
        slide_out := jsonb_set(slide_out, ARRAY[arr_field], arr_out);
      END IF;
    END LOOP;
    -- Watermark singleton (object, not array).
    IF slide_out ? 'watermark' AND jsonb_typeof(slide_out -> 'watermark') = 'object' THEN
      slide_out := jsonb_set(slide_out, '{watermark}', (slide_out -> 'watermark') #- '{locked}');
    END IF;
    result := result || jsonb_build_array(slide_out);
  END LOOP;
  RETURN result;
END;
$$;
```

      ```typescript
      import type { DecorationContent } from './schemas'
      import type { Slide } from '@/lib/types/slide'

      export const DEFAULT_MONOCULTURE_THRESHOLD = 0.60

      export interface DiversityCheckResult {
        warned: boolean
        dominantKind?: 'icon' | 'shape'
        dominantName?: string  // lucideName when dominantKind === 'icon'
        dominantShapeKind?: string  // when dominantKind === 'shape'
        dominantCount?: number
        total?: number
        share?: number  // 0..1
      }

      export interface DiversityCheckOpts {
        threshold?: number  // default DEFAULT_MONOCULTURE_THRESHOLD
        carouselId?: string  // for warn message
      }

      function isIconDecoration(d: unknown): d is { kind: 'icon'; lucideName: string } { ... }
      function isShapeDecoration(d: unknown): d is { kind: 'shape'; shapeKind: string } { ... }

      export function decorationDiversityCheck(
        slides: ReadonlyArray<Pick<Slide, 'decorations'>>,
        opts: DiversityCheckOpts = {},
      ): DiversityCheckResult {
        const threshold = opts.threshold ?? DEFAULT_MONOCULTURE_THRESHOLD
        const iconFreq = new Map<string, number>()
        const shapeFreq = new Map<string, number>()
        for (const slide of slides) {
          if (!Array.isArray(slide.decorations)) continue
          for (const d of slide.decorations) {
            if (isIconDecoration(d)) {
              iconFreq.set(d.lucideName, (iconFreq.get(d.lucideName) ?? 0) + 1)
            } else if (isShapeDecoration(d)) {
              shapeFreq.set(d.shapeKind, (shapeFreq.get(d.shapeKind) ?? 0) + 1)
            }
          }
        }
        // Compute per-population dominance
        function dominance(freq: Map<string, number>): { name: string; count: number; total: number; share: number } | null {
          let total = 0
          let topName = ''
          let topCount = 0
          for (const [name, count] of freq) {
            total += count
            if (count > topCount) { topCount = count; topName = name }
          }
          if (total === 0) return null
          return { name: topName, count: topCount, total, share: topCount / total }
        }
        const iconDom = dominance(iconFreq)
        const shapeDom = dominance(shapeFreq)
        const candidates = [
          iconDom ? { kind: 'icon' as const, ...iconDom } : null,
          shapeDom ? { kind: 'shape' as const, ...shapeDom } : null,
        ].filter((c): c is NonNullable<typeof c> => c !== null)
        const triggered = candidates.find(c => c.share >= threshold)
        if (!triggered) return { warned: false }
        const carouselTag = opts.carouselId ? ` carousel=${opts.carouselId}` : ''
        const labelKey = triggered.kind === 'icon' ? 'lucideName' : 'shapeKind'
        console.warn(
          `[decoration-diversity]${carouselTag} ${labelKey}='${triggered.name}' dominance ${(triggered.share*100).toFixed(1)}% (${triggered.count}/${triggered.total}) >= threshold ${(threshold*100).toFixed(0)}% — ADVISORY (does not fail CI in v2.12)`,
        )
        return triggered.kind === 'icon'
          ? { warned: true, dominantKind: 'icon', dominantName: triggered.name, dominantCount: triggered.count, total: triggered.total, share: triggered.share }
          : { warned: true, dominantKind: 'shape', dominantShapeKind: triggered.name, dominantCount: triggered.count, total: triggered.total, share: triggered.share }
      }
      ```
