```typescript
// Phase 67 (DECOR-01..04): Decoration types — discriminated union for icons + shapes.
// Schema is purely additive on Slide; both renderers iterate (slide.decorations ?? [])
// at layer 5.5 (between custom_images at 5.0 and watermark at 6.0). When the
// array is undefined, zero DOM nodes are emitted — byte-identical to pre-v2.12
// baseline. Actual rendering deferred to Phase 69 (decoration-renderer.tsx).
//
// Color fields are ALWAYS explicit hex (e.g. '#FF0000') — never 'currentColor',
// never CSS named colors, never CSS variables. PCL12 grep gate (Phase 69)
// enforces this in src/lib/render/. The Phase 71 AI-ingest pipeline will
// resolve named-token colors ('accent' / 'palette-vibrant') to hex server-side
// BEFORE any decoration reaches this type.

export type DecorationShadowEffect = {
  enabled: boolean
  color: string          // 8-digit hex e.g. '#00000080' (mirror v2.1 TextShadowEffect)
  size: 'sm' | 'md' | 'lg'
}

export interface DecorationEffects {
  shadow?: DecorationShadowEffect
}

export interface DecorationBase {
  id: string                       // crypto.randomUUID() — same as TextBox/CustomImage
  /** Position + size as % of canvas (0-100). 4-tuple matches CustomImage layout
      semantics. Both renderers translate to px at REF dimensions (1080x1920). */
  position: { x: number; y: number; width: number; height: number }
  /** Multi-select meta-fields — same shape as TextBox + CustomImage + Watermark.
      Reducer support extends in Phase 68. */
  groupId?: string
  locked?: boolean
  hidden?: boolean
  /** User-supplied label override for the layers panel. Mirrors LAYERS-04. */
  name?: string
  /** Rotation in degrees, default 0. CSS transform: rotate() at render time
      (Phase 69 wires renderer; Phase 72 wires interactive handle). */
  rotation?: number
  /** Opacity 0..1, default 1. CSS opacity at render time. */
  opacity?: number
  /** Effects (drop shadow only this milestone). CSS box-shadow on wrapping
      <div> at render time — never SVG <filter> (vercel/satori#573). */
  effects?: DecorationEffects
}

export interface IconDecoration extends DecorationBase {
  kind: 'icon'
  /** Closed-enum at AI-ingest boundary; OPEN in this type (UI picker emits
      arbitrary strings from the lucide registry; the Zod validator at the
      AI boundary in Phase 71 silently drops unknowns). */
  lucideName: string
  /** Explicit hex (e.g. '#1F2937'). Never 'currentColor' (vercel/satori#41). */
  color: string
  /** Stroke width in px (typical Lucide range 1-3). */
  strokeWidth: number
}

export type ShapeKind =
  | 'rect'
  | 'circle'
  | 'ellipse'
  | 'triangle'
  | 'polygon'
  | 'star'
  | 'arrow'
  | 'line'

export interface DecorationStroke {
  color: string                    // explicit hex
  width: number                    // px
  style: 'solid' | 'dashed' | 'dotted'
}

export interface ShapeDecoration extends DecorationBase {
  kind: 'shape'
  shapeKind: ShapeKind
  /** Explicit hex fill (e.g. '#A8E0B0'). */
  fill: string
  stroke?: DecorationStroke
  /** Corner radius in px — only meaningful for shapeKind: 'rect'. Renderer
      ignores on other kinds. */
  cornerRadius?: number
  /** Vertex count — only meaningful for shapeKind: 'polygon' (5/6/8) or
      'star' (5/6 points). Renderer ignores on other kinds. */
  points?: number
}

export type Decoration = IconDecoration | ShapeDecoration
```

// Phase 67 (DECOR-05): Decoration discriminated-union Zod schema. Registered
// at the AI-ingest boundary; Phase 71 will wire Gemini emit to consume this.
// Plan 67-01 lands the schema only — no runtime consumer yet.
//
// Silent-drop pattern (mirrors v2.9 D-08 template_id_override): unknown
// lucideName (or any other malformed entry) is dropped from the array via
// parseDecorationsArraySafe — never throws. The closed-enum lucideName
// allowlist is a Phase 71 concern; this layer accepts any non-empty string.

export const DecorationPositionSchema = z.object({
  x: z.number(),
  y: z.number(),
  width: z.number(),
  height: z.number(),
})

export const DecorationStrokeSchema = z.object({
  color: z.string(),
  width: z.number(),
  style: z.enum(['solid', 'dashed', 'dotted']),
})

export const DecorationShadowEffectSchema = z.object({
  enabled: z.boolean(),
  color: z.string(),
  size: z.enum(['sm', 'md', 'lg']),
})

export const DecorationEffectsSchema = z.object({
  shadow: DecorationShadowEffectSchema.optional(),
})

const DecorationBaseFields = {
  id: z.string(),
  position: DecorationPositionSchema,
  groupId: z.string().optional(),
  locked: z.boolean().optional(),
  hidden: z.boolean().optional(),
  name: z.string().optional(),
  rotation: z.number().optional(),
  opacity: z.number().optional(),
  effects: DecorationEffectsSchema.optional(),
} as const

  lucideName: z.string().min(1),
  color: z.string(),
  strokeWidth: z.number(),
})

  kind: z.literal('shape'),
  ...DecorationBaseFields,
  shapeKind: z.enum(['rect', 'circle', 'ellipse', 'triangle', 'polygon', 'star', 'arrow', 'line']),
  fill: z.string(),
  stroke: DecorationStrokeSchema.optional(),
  cornerRadius: z.number().optional(),
  points: z.number().optional(),
})

export const DecorationSchema = z.discriminatedUnion('kind', [
  IconDecorationSchema,
  ShapeDecorationSchema,
])

/**
 * Phase 67 (DECOR-05): silent-drop array parser. Per-entry safeParse; malformed
 * entries are filtered without throwing. Mirrors v2.9 D-08 template_id_override
 * defaulting pattern. Phase 71 will additionally clamp to maxItems: 2 and
 * gate lucideName against the closed-enum allowlist — both layered ON TOP of
 * this base validator.
 */
export function parseDecorationsArraySafe(input: unknown): import('zod').infer<typeof DecorationSchema>[] {
  if (!Array.isArray(input)) return []
  const out: import('zod').infer<typeof DecorationSchema>[] = []
  for (const entry of input) {
    const result = DecorationSchema.safeParse(entry)
    if (result.success) out.push(result.data)
    // else: silent drop — never throws (DECOR-05)
  }
  return out
}

export type DecorationContent = z.infer<typeof DecorationSchema>
export type IconDecorationContent = z.infer<typeof IconDecorationSchema>
export type ShapeDecorationContent = z.infer<typeof ShapeDecorationSchema>

function grepImports(needle: string): string[] {
  // grep -rE 'from .lucide-react.' src/ — narrow to lucide-react import lines first,
  // then filter for the icon name. Use try/catch since grep exits 1 on no matches.
  try {
    const out = execSync(
      `grep -rE "from ['\\\"]lucide-react['\\\"]" "${SRC_DIR}" --include="*.ts" --include="*.tsx" -l`,
      { encoding: 'utf8' },
    )
    const files = out.split('\n').filter(Boolean)
    const offenders: string[] = []
    for (const f of files) {
      const content = readFileSync(f, 'utf8')
      // Find the lucide-react import block(s) and check for the needle.
      // Match patterns like:  import { A, B, Github, C } from 'lucide-react'
      // (multi-line imports also covered by checking the file content + a regex for the needle within the import block).
      const importBlocks = content.match(/import\s*\{[^}]*\}\s*from\s*['"]lucide-react['"]/gs) ?? []
      for (const block of importBlocks) {
        const re = new RegExp(`\\b${needle}\\b`)
        if (re.test(block)) {
          offenders.push(`${f}: imports ${needle}`)
        }
      }
    }
    return offenders
  } catch (err: unknown) {
    // grep exits 1 when no matches — that's a clean state for our gate.
    const stderr = err && typeof err === 'object' && 'stderr' in err ? String((err as { stderr: unknown }).stderr) : ''
    if (stderr.includes('No such file')) throw err
    return []
  }
}
