       export const DEFAULT_ICON_STROKE_WIDTH = 2  // Lucide default

       export function buildDefaultIconDecoration(
         lucideName: string,
         opts?: { center?: { x: number; y: number } }
       ): IconDecoration | null {
         if (!isAllowedLucideName(lucideName)) return null  // closed-enum guard (defensive)
         const center = opts?.center ?? { x: 50, y: 50 }
         const w = DEFAULT_ICON_SIZE_PCT.width

         const x = clamp(center.x - w / 2, 0, 100 - w)
         const y = clamp(center.y - h / 2, 0, 100 - h)

           kind: 'icon',

         }
       }

    2. lucide-picker-corpus.ts (~80 LOC):
       ```ts
       import { LUCIDE_DECORATION_ALLOWLIST } from '@/lib/render/decorations/registry'
       import uFuzzy from '@leeoniya/ufuzzy'

       export interface PickerCorpusEntry {
         lucideName: string         // PascalCase canonical
         displayName: string        // PascalCase → space-separated for UI ("ArrowRight" → "Arrow Right")
         searchTokens: string       // lowercase searchable string
       }

       function pascalCaseToWords(name: string): string {
         return name.replace(/([A-Z])/g, ' $1').trim()
       }

       export const LUCIDE_PICKER_CORPUS: readonly PickerCorpusEntry[] =
         LUCIDE_DECORATION_ALLOWLIST.map((name) => ({
           lucideName: name,
           displayName: pascalCaseToWords(name),
           searchTokens: `${name.toLowerCase()} ${pascalCaseToWords(name).toLowerCase()}`,
         }))

       const SEARCH_HAYSTACK: readonly string[] = LUCIDE_PICKER_CORPUS.map((e) => e.searchTokens)

       // ufuzzy options: case-insensitive (lowered haystack), fuzzy with 1 typo allowed,
       // grouped output for stable result order. Keep options in a const so tests can re-run
       // deterministically.
       const FUZZY_OPTS = { intraIns: 1 } // allow 1 inserted char between matched chars
       const fuzzy = new uFuzzy(FUZZY_OPTS)

       export function filterIcons(query: string, corpus = LUCIDE_PICKER_CORPUS): PickerCorpusEntry[] {
         if (!query.trim()) return corpus.slice() // empty query = full list
         const haystack = SEARCH_HAYSTACK as string[]
         const needles = [query.toLowerCase()]
         const [idxs] = fuzzy.search(haystack, needles[0])
         if (!idxs) return []
         return idxs.map((i) => corpus[i])
       }

       export function createUfuzzy(): uFuzzy {
         return new uFuzzy(FUZZY_OPTS)
       }
       ```
