    Implementation:
    ```typescript
    const DEFAULT_MODEL = 'gemini-2.5-flash'

    export async function lookupStyleSpec(admin, sha256) {
      try {
        const { data, error } = await admin
          .from('style_spec_cache')
          .select('style_spec')
          .eq('sha256', sha256)
          .maybeSingle()
        if (error || !data) return null
        // Fire-and-forget last_hit_at bump — never await; never throw
        admin
          .from('style_spec_cache')
          .update({ last_hit_at: new Date().toISOString() })
          .eq('sha256', sha256)
          .then(({ error: updateErr }) => {
            if (updateErr) console.warn('[style-spec-cache] last_hit_at update failed:', updateErr)
          })
        return data.style_spec
      } catch (err) {
        console.warn('[style-spec-cache] lookup failed; treating as miss:', err)
        return null
      }
    }

    export async function persistStyleSpec(admin, sha256, styleSpec, model = DEFAULT_MODEL) {
      try {
        const { error } = await admin
          .from('style_spec_cache')
          .upsert(
            { sha256, style_spec: styleSpec, model },
            { onConflict: 'sha256' },
          )
        if (error) console.warn('[style-spec-cache] persist failed (non-blocking):', error)
      } catch (err) {
        console.warn('[style-spec-cache] persist threw (non-blocking):', err)
      }
    }
    ```
