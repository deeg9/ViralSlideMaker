    ```tsx
    'use client'
    import type { RefusalFlags } from '@/lib/ai/refusal-pre-pass'
    import type { SpecPreviewAction } from './state-machine'
    export interface RefusalWarningProps {
      flags: RefusalFlags
      acknowledged: boolean
      dispatch: React.Dispatch<SpecPreviewAction>
    }
    export function RefusalWarning({ flags, acknowledged, dispatch }: RefusalWarningProps) {
      const hits: string[] = []
      if (flags.contains_logo) hits.push('logo')
      if (flags.contains_face) hits.push('face')
      if (flags.contains_watermark) hits.push('watermark')
      if (hits.length === 0) return null

      const subjects = hits.join(' + ')
      // PREVIEW-10 + LEGAL-03 verbatim copy:
      const message = `This reference contains a ${subjects} — generated output will be inspired by the style, not the brand identity.`

      return (
        <div role="alert" aria-live="polite" data-testid="refusal-warning"
             className="text-xs bg-[#fff8e1] border border-[#f0c674] rounded-md px-2 py-1 mt-1">
          <p>{message}</p>
          {acknowledged ? (
            <span className="text-[10px] text-[#7d6161]">Acknowledged ✓</span>
          ) : (
            <button
              type="button"
              onClick={() => dispatch({ type: 'CONFIRM_REFUSAL_WARNING' })}
              data-testid="refusal-continue-cta"
              className="text-[11px] font-bold text-[#3d0000] underline"
            >Continue</button>
          )}
        </div>
      )
    }
    ```

    ```tsx
    import type { SpecPreviewState, SpecPreviewAction } from './state-machine'
    import { SafetyBlockRecovery } from './safety-block-recovery'

    export interface ErrorRouterProps {
      state: SpecPreviewState
      dispatch: React.Dispatch<SpecPreviewAction>
    }

    export function ErrorRouter({ state, dispatch }: ErrorRouterProps) {
      if (!state.error) return null
      switch (state.error.kind) {
        case 'safety_blocked':
          return <SafetyBlockRecovery recoverySuggestion={state.error.recovery_suggestion} dispatch={dispatch} />
        case 'cap_reached':
          return (
            <section data-testid="error-cap-reached" role="region" className="...">
              <p>You&apos;ve reached today&apos;s extraction limit (5 references/day on free tier).</p>
              <Link href="/pricing" data-testid="upgrade-link">Upgrade</Link>
            </section>
          )
        case 'insufficient_credits':
          return (
            <section data-testid="error-insufficient-credits" role="region" className="...">
              <p>Out of credits.</p>
              <Link href="/pricing" data-testid="get-credits-link">Get credits</Link>
            </section>
          )
        case 'extraction_failed':
          return (
            <section data-testid="error-extraction-failed" role="region" className="...">
              <p>We couldn&apos;t extract a style from these references.</p>
              <button onClick={() => dispatch({ type: 'CLEAR_ERROR' })} data-testid="try-again-cta">Try again</button>
            </section>
          )
        case 'no-retries-left':
          return (
            <section data-testid="error-no-retries" role="region" className="...">
              <p>Free retries used. Upload a new reference set to try again.</p>
              <button onClick={() => dispatch({ type: 'UPLOAD_RESET', references: [] })} data-testid="reset-uploads-cta">Start over</button>
            </section>
          )
        case 'preview-required':
        case 'no-references':
          return (
            <p data-testid="error-inline-hint" role="alert" className="text-xs text-[#a30000]">
              {state.error.kind === 'no-references'

        default: {
          const _exhaustive: never = state.error.kind
          return null
        }
      }
    }
    ```
