```typescript
interface StyleSpec {
  schema_version: 1
  palette: { mode: 'solid' | 'duotone' | 'gradient' | 'photo'; primary: string; secondary?: string; accent?: string }
  typography: { hookFontCategory?: 'sans-serif' | 'serif' | 'display' | 'handwriting'; bodyFontCategory?: same }
  layout: { hookPosition?: 'top' | 'center' | 'bottom'; density?: 'tight' | 'medium' | 'airy' }
  imagery: { treatmentPreset?: 'bw' | 'duotone-cream' | 'duotone-cobalt' | 'contrast-bump' }
  decorations: { preferredLucide?: string[]; preferredShapes?: string[]; density?: 'sparse' | 'moderate' | 'dense' }
  rhythm: { cadence?: 'punchy' | 'flowing' | 'sparse' }
  voice_note?: string  // sanitized HTML-stripped, ≤200 chars, single line
}
```

interface BrandKit {
  primary_color: string | null; secondary_color: string | null; accent_color: string | null

  heading_font_family: string | null; body_font_family: string | null
  // ...other fields
}

    'use client'
    import type { SpecPreviewState, SpecPreviewAction } from './state-machine'
    export interface PerAspectTogglesProps {
      aspects: SpecPreviewState['aspects']
      dispatch: React.Dispatch<SpecPreviewAction>
    }
    export function PerAspectToggles({ aspects, dispatch }: PerAspectTogglesProps) {
      const items: Array<{ key: keyof SpecPreviewState['aspects']; label: string }> = [
        { key: 'palette', label: 'Palette' },
        { key: 'typography', label: 'Typography' },
        { key: 'layout', label: 'Layout' },
        { key: 'decorations', label: 'Decorations' },
      ]
      const noneChecked = !items.some(i => aspects[i.key])
      return (
        <fieldset aria-label="Choose which extracted aspects to apply">
          {items.map(item => (
            <label key={item.key} htmlFor={`aspect-${item.key}`}>
              <input
                id={`aspect-${item.key}`}
                type="checkbox"
                checked={aspects[item.key]}
                onChange={() => dispatch({ type: 'TOGGLE_ASPECT', aspect: item.key })}
              />
              {item.label}
            </label>
          ))}
          {noneChecked && (
            <p data-testid="aspect-empty-hint" className="text-xs text-[#a30000]">
              Select at least one aspect to generate.
            </p>
          )}
        </fieldset>
      )
    }
