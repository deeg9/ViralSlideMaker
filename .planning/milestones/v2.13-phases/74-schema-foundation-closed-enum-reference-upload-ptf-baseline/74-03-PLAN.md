    // Phase 74 Plan 03 (REF-02): signed-URL upload route for the slide-references bucket.
    // Mirrors src/app/api/v1/me/brand-kit/logo-upload-url/route.ts shape with these
    // differences:
    //   - Bucket: SLIDE_REFERENCES_BUCKET (PRIVATE — no publicUrl in response)
    //   - Allowed MIME: ALLOWED_REFERENCE_MIME (no svg+xml — vision LLM scope)
    //   - Adds server-side size validation (REF-05, ≤ 20MB pre-preprocessing)
    //   - Returns `path` only (no publicUrl); Phase 75 mints signed READ URLs at
    //     extraction time.

    import { withApiAuth } from '@/lib/api/with-api-auth'
    import { apiSuccess } from '@/lib/api/response'
    import { apiError } from '@/lib/api/errors'
    import { SLIDE_REFERENCES_BUCKET } from '@/lib/types/projects-storage-constants'
    import { validateReferenceUpload } from '@/lib/storage/upload-reference-validation'

    export const POST = withApiAuth(async (req, ctx) => {
      let body: Record<string, unknown> = {}
      try {
        const text = await req.text()
        if (text.length > 0) body = JSON.parse(text)
      } catch {
        return apiError('VALIDATION_ERROR', 'Invalid JSON body', 400)
      }

      const validation = validateReferenceUpload({
        filename: body.filename,
        contentType: body.contentType,
        size_bytes: body.size_bytes,
      })
      if (!validation.ok) return apiError('VALIDATION_ERROR', validation.message, 400)

      const path = `${ctx.userId}/${crypto.randomUUID()}-${validation.safeFilename}`

      const { data, error } = await ctx.supabase.storage
        .from(SLIDE_REFERENCES_BUCKET)
        .createSignedUploadUrl(path)
      if (error || !data) {
        return apiError('INTERNAL_ERROR', error?.message ?? 'Failed to create signed upload URL', 500)
      }

      return apiSuccess({
        data: {
          uploadUrl: data.signedUrl,
          token: data.token,
          method: 'PUT' as const,
          headers: { 'Content-Type': String(body.contentType) },
          path,
        },
      })
    })
    ```

    'use client'
    // Phase 74 Plan 03 (REF-02..REF-06): reference image uploader.
    // Native HTML5 drag-drop (no react-dropzone — confirmed absent from package.json).
    // Mirrors BrandKitLogoUploader naming convention (per CONTEXT.md Claude's-discretion).
    //
    // REF-06 SEPARATION: upload completion fires onUploadComplete(row); user
    // explicitly clicks "Extract style" to fire onExtractRequested(rows).
    // This component does NOT itself call vision — Phase 75 owns extraction.

    import { useState, useRef, useCallback } from 'react'
    import {
      validateReferenceUpload,
      validateGalleryCount,
      MAX_REFERENCE_COUNT,
    } from '@/lib/storage/upload-reference-validation'
    import type { ReferenceAssetRow } from '@/lib/types/projects-storage-constants'

    export interface ReferenceImageUploaderProps {
      onUploadComplete?: (row: ReferenceAssetRow) => void
      onExtractRequested?: (rows: ReferenceAssetRow[]) => void
      initial?: ReferenceAssetRow[]
    }

    type Thumb = ReferenceAssetRow & { _objectUrl: string }

    export function ReferenceImageUploader(props: ReferenceImageUploaderProps) {
      const [thumbs, setThumbs] = useState<Thumb[]>(/* derive from initial — without _objectUrl when remote */)
      const [error, setError] = useState<string | null>(null)
      const [isDragOver, setIsDragOver] = useState(false)
      const fileInputRef = useRef<HTMLInputElement>(null)

      const handleFiles = useCallback(async (files: File[]) => {
        // 1. Gallery count validation (REF-03)
        const countCheck = validateGalleryCount(thumbs.length, files.length)
        if (!countCheck.ok) { setError(countCheck.message); return }

        // 2. Per-file validation + signed-URL upload (REF-04, REF-05)
        for (const file of files) {
          const v = validateReferenceUpload({
            filename: file.name,
            contentType: file.type,
            size_bytes: file.size,
          })
          if (!v.ok) { setError(v.message); return }

          // 2a. Mint signed URL
          const urlRes = await fetch('/api/style-references/upload-url', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ filename: file.name, contentType: file.type, size_bytes: file.size }),
          })
          if (!urlRes.ok) { setError('Failed to mint upload URL.'); return }
          const { data } = await urlRes.json()

          // 2b. PUT direct to Storage (REF-02 — bypasses 4MB Vercel cap)
          const putRes = await fetch(data.uploadUrl, {
            method: data.method,
            headers: data.headers,
            body: file,
          })
          if (!putRes.ok) { setError('Upload failed. Please try again.'); return }

          // 2c. Build ReferenceAssetRow + add thumbnail (REF-07 provenance row)
          const objectUrl = URL.createObjectURL(file)
          const row: ReferenceAssetRow = {
            url: data.path,                // store path; Phase 75 mints signed READ URL on demand
            sha256: '',                    // placeholder; Phase 73 prepareForVision computes the real SHA at extraction time
            uploaded_at: new Date().toISOString(),
            mime_type: file.type as ReferenceAssetRow['mime_type'],
            size_bytes: file.size,
            original_filename: file.name,
          }
          setThumbs((prev) => [...prev, { ...row, _objectUrl: objectUrl }])
          props.onUploadComplete?.(row)
        }
      }, [thumbs.length, props])

      const handleRemove = (idx: number) => {
        setThumbs((prev) => {
          URL.revokeObjectURL(prev[idx]._objectUrl)
          return prev.filter((_, i) => i !== idx)
        })
      }

      const handleExtract = () => {
        const rows: ReferenceAssetRow[] = thumbs.map(({ _objectUrl, ...row }) => row)
        props.onExtractRequested?.(rows)
      }

      // Native HTML5 drag-drop handlers
      const onDragOver = (e: React.DragEvent) => { e.preventDefault(); setIsDragOver(true) }
      const onDragLeave = () => setIsDragOver(false)
      const onDrop = (e: React.DragEvent) => {
        e.preventDefault()
        setIsDragOver(false)
        const files = Array.from(e.dataTransfer.files)
        void handleFiles(files)
      }

      return (
        <div>
          <div
            data-testid="reference-uploader-dropzone"
            onClick={() => fileInputRef.current?.click()}
            onDragOver={onDragOver}
            onDragLeave={onDragLeave}
            onDrop={onDrop}
            className={isDragOver ? 'active' : ''}
          >
            {/* drop zone copy */}
          </div>
          <input
            ref={fileInputRef}
            type="file"
            multiple
            accept="image/png,image/jpeg,image/webp,image/heic"
            onChange={(e) => void handleFiles(Array.from(e.target.files ?? []))}
            style={{ display: 'none' }}
          />
          {error && <div role="alert" data-testid="uploader-error">{error}</div>}
          <ul data-testid="thumb-row">
            {thumbs.map((t, i) => (
              <li key={i}>
                <img src={t._objectUrl || t.url} alt={t.original_filename} />
                <button onClick={() => handleRemove(i)} aria-label={`Remove ${t.original_filename}`}>X</button>
              </li>
            ))}
          </ul>
          <button
            data-testid="extract-style-cta"
            disabled={thumbs.length < 1 || thumbs.length > MAX_REFERENCE_COUNT}
            onClick={handleExtract}
          >
            Extract style
          </button>
        </div>
      )
    }
    ```
