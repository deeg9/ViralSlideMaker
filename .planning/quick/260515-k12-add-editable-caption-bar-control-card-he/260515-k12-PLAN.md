   ```typescript
   /** quick-260515-k12: compose an rgba() string from a hex color + 0..1 alpha.
       Accepts #RGB / #RRGGBB. Invalid input falls back to rgba(0,0,0,alpha). */
   export function hexToRgba(hex: string, alpha: number): string {
     let h = hex.replace('#', '')
     if (h.length === 3) h = h.split('').map((c) => c + c).join('')
     const r = parseInt(h.slice(0, 2), 16)
     const g = parseInt(h.slice(2, 4), 16)
     const b = parseInt(h.slice(4, 6), 16)
     if (Number.isNaN(r) || Number.isNaN(g) || Number.isNaN(b)) {
       return `rgba(0, 0, 0, ${alpha})`
     }
     return `rgba(${r}, ${g}, ${b}, ${alpha})`
   }

   export interface ResolvedCaptionBar {
     heightPct: number          // 10..90
     backgroundColor: string    // rgba() string
     position: 'bottom' | 'top'
   }

   /** quick-260515-k12: SINGLE source of truth for caption-bar geometry/color.
       Both slide-preview.tsx and render-slide-satori.tsx import this so the
       browser preview and the Satori PNG export cannot drift (PARITY-01).

       Byte-identity (TPLREND-05): with NO caption-bar overrides the result is
       { heightPct: 38, backgroundColor: <bottomCardBackground(template.backgroundColor)>,
         position: 'bottom' } — exactly what the old hardcoded path produced. */
   export function resolveCaptionBar(
     overrides: SlideStyleOverrides | undefined,
     template: SlideTemplate,
   ): ResolvedCaptionBar {
     const o = overrides ?? {}
     const heightPct = clamp(o.caption_bar_height_pct ?? 38, 10, 90)
     // Legacy path: when BOTH color and opacity are unset, reuse bottomCardBackground
     // verbatim so an 8-digit-hex template.backgroundColor still passes through and a
     // 6-digit one still yields the baked 0.85 — guarantees byte-identity.
     const hasColor = o.caption_bar_color != null
     const hasOpacity = o.caption_bar_opacity != null
     const backgroundColor =
       !hasColor && !hasOpacity
         ? bottomCardBackground(template.backgroundColor)
         : hexToRgba(
             o.caption_bar_color ?? template.backgroundColor,
             o.caption_bar_opacity ?? 0.85,
           )
     const position = o.caption_bar_position ?? 'bottom'
     return { heightPct, backgroundColor, position }
   }
   ```

| Boundary | Description |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-k12-01 | Tampering | `caption_bar_height_pct` could be set out of range via a hand-edited project payload | mitigate | `resolveCaptionBar` clamps height to 10..90 before render — no unbounded value reaches the layout. |
| T-k12-02 | Tampering | Malformed `caption_bar_color` hex could break the rgba() string | mitigate | `hexToRgba` validates parsed RGB; `NaN` falls back to `rgba(0, 0, 0, alpha)` — render never throws. |
| T-k12-03 | Information disclosure | Caption-bar card leaking into the published/exported render path | accept | The card only writes data; renderers read data. Editor chrome is not added to the render tree — no `editable` leak surface. No PII involved. |
</threat_model>
