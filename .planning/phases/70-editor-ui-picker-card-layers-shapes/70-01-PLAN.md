export type ShapeKind =
  | 'rect' | 'circle' | 'ellipse'
  | 'triangle' | 'polygon' | 'star'
  | 'arrow' | 'line'

export interface DecorationStroke {
  color: string                    // explicit hex
  width: number                    // px
  style: 'solid' | 'dashed' | 'dotted'
}

export interface DecorationBase {
  id: string                       // crypto.randomUUID()
  position: { x: number; y: number; width: number; height: number }  // % of canvas
  groupId?: string
  locked?: boolean
  hidden?: boolean
  name?: string
  rotation?: number                // degrees, default 0 — Phase 72 owns interactive
  opacity?: number                 // 0..1 (NOT 0..100; CSS-native scale, WR-01)
  effects?: DecorationEffects
}

export interface ShapeDecoration extends DecorationBase {
  kind: 'shape'
  shapeKind: ShapeKind
  fill: string                     // explicit hex
  stroke?: DecorationStroke
  cornerRadius?: number            // rect only; renderer ignores on others
  points?: number                  // polygon (5/6/8) or star (5/6) — renderer ignores on others
}

export type Decoration = IconDecoration | ShapeDecoration

         const size = shapeKind === 'line' ? DEFAULT_LINE_SIZE_PCT : DEFAULT_SHAPE_SIZE_PCT
         const x = clamp(center.x - size.width / 2, 0, 100 - size.width)
