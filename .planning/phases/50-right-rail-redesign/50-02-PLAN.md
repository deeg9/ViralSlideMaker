Composition (top-to-bottom):

```tsx
export function TextEditorPanel(props: TextEditorPanelProps) {
  const { slide, template, selectedTextBoxId, onTextBoxAction, onUpdate, ...rest } = props
  const textBoxes = slide.text_boxes ?? []
  const activeTextBox = textBoxes.find((tb) => tb.id === selectedTextBoxId) ?? textBoxes[0] ?? null

  const updateActiveTextBox = (updates: Partial<TextBox>) => {
    if (!activeTextBox) return
    onTextBoxAction({
      type: 'UPDATE_TEXT_BOX',
      slideIndex: -1,  // resolved by parent
      textBoxId: activeTextBox.id,
      updates,
    })
  }

  return (
    <div className="flex flex-col gap-4 p-6" data-testid="text-editor-panel">
      <LayerStrip
        textBoxes={textBoxes}
        selectedTextBoxId={selectedTextBoxId}
        onSelect={(id) => onTextBoxAction({ type: 'SET_ACTIVE_TEXT_BOX_ID', slideIndex: -1, textBoxId: id } as any /* resolved at slide-editor */)}
        onAdd={() => onTextBoxAction({ type: 'ADD_TEXT_BOX', slideIndex: -1, role: 'custom' })}
      />

      {activeTextBox && (
        <ActiveTextBoxCard
          textBox={activeTextBox}
          onUpdate={updateActiveTextBox}
          template={template}
          brandKit={rest.brandKit}
        />
      )}

      <SlideSettings
        slide={slide}
        template={template}
        onUpdate={onUpdate}
        brandKit={rest.brandKit}
      />

      {/* Existing custom-text editor section preserved at bottom — handles selection of free-positioned text boxes from canvas (Phase 52 will reduce this surface) */}
      {rest.selectedCustomTextId && rest.customTexts && rest.onCustomTextUpdate && rest.onCustomTextRemove && (
        <CustomTextEditor
          customText={rest.customTexts.find((ct) => ct.id === rest.selectedCustomTextId)}
          onUpdate={(updates) => rest.onCustomTextUpdate!(rest.selectedCustomTextId!, updates)}
          onRemove={() => rest.onCustomTextRemove!(rest.selectedCustomTextId!)}
        />
      )}
    </div>
  )
}
```

Note on the SET_ACTIVE_TEXT_BOX_ID action: per Phase 49 D-16, selection lives OUTSIDE the reducer in React state. The cleanest contract is to use a separate prop `onSelectTextBox: (id: string | null) => void` instead of routing through the action union. Update prop interface:

```typescript
onSelectTextBox: (textBoxId: string | null) => void  // calls setSelectedTextBoxId from useSlideEditor
```

Replace `onTextBoxAction` with two narrower props:
- `onTextBoxAction: (action: { type: 'ADD_TEXT_BOX' | 'UPDATE_TEXT_BOX' | 'REMOVE_TEXT_BOX', ... }) => void`
- `onSelectTextBox: (textBoxId: string | null) => void`

**Acceptance:** New panel composes 5 children (LayerStrip, ActiveTextBoxCard, SlideSettings, optional CustomTextEditor); typecheck passes.

### T-03 — Wire into `slide-editor.tsx`

**File:** `src/components/editor/slide-editor.tsx`
**Action:** Edit at the panel mount site (~line 386)

Read the existing `useSlideEditor` hook return — Phase 49 added `selectedTextBoxId`, `setSelectedTextBoxId`, `removeTextBox`. Wire them through:

```tsx
import { TextEditorPanel } from './text-editor-panel'  // already imports
import { LegacyTextEditorPanel } from './text-editor-panel-legacy'

const useNewRail = process.env.NEXT_PUBLIC_NEW_TEXT_RAIL !== 'false'

const PanelComponent = useNewRail ? TextEditorPanel : LegacyTextEditorPanel

<PanelComponent
  slide={currentSlide}
  template={template}
  selectedTextBoxId={selectedTextBoxId}
  onSelectTextBox={setSelectedTextBoxId}
  onTextBoxAction={(action) => {
    // resolve slideIndex from current selection
    dispatch({ ...action, slideIndex: selectedIndex } as EditorAction)
  }}
  onUpdate={(updates) => dispatch({ type: 'UPDATE_SLIDE', index: selectedIndex, updates })}
  selectedCustomTextId={selectedCustomTextId}
  customTexts={currentSlide.custom_texts}
  onCustomTextUpdate={(id, updates) => dispatch({ type: 'UPDATE_CUSTOM_TEXT', slideIndex: selectedIndex, textId: id, updates })}
  onCustomTextRemove={(id) => dispatch({ type: 'REMOVE_CUSTOM_TEXT', slideIndex: selectedIndex, textId: id })}
  brandKit={brandKit}
/>
```

If `LegacyTextEditorPanel` doesn't accept the new props, gate the prop passing with the flag:

```tsx
{useNewRail ? (
  <TextEditorPanel {...newProps} />
) : (
  <LegacyTextEditorPanel {...legacyProps} />
)}
```

**Acceptance:** Either `<TextEditorPanel>` or `<LegacyTextEditorPanel>` mounts based on env flag; no TypeScript errors.
