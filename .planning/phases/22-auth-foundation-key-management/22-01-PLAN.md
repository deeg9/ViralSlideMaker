```typescript
import { createHash, timingSafeEqual } from 'crypto'
import { createAdminClient } from '@/lib/supabase/admin'
import { NextRequest } from 'next/server'

export async function authenticateApiKey(
  request: NextRequest
): Promise<{ userId: string; keyId: string }> {
  const authHeader = request.headers.get('authorization')
  if (!authHeader?.startsWith('Bearer vsl_')) {
    throw new Error('Missing or invalid Authorization header')
  }

  const rawKey = authHeader.slice(7) // Remove "Bearer "
  const keyHash = createHash('sha256').update(rawKey).digest('hex')

  const admin = createAdminClient()
  const { data, error } = await admin
    .from('api_keys')
    .select('id, user_id, key_hash')
    .eq('key_hash', keyHash)
    .single()

  if (error || !data) {
    throw new Error('Invalid API key')
  }

  // Timing-safe comparison as defense in depth (per AUTH-02)
  const storedHashBuf = Buffer.from(data.key_hash, 'utf8')
  const computedHashBuf = Buffer.from(keyHash, 'utf8')
  if (!timingSafeEqual(storedHashBuf, computedHashBuf)) {
    throw new Error('Invalid API key')
  }

  // Fire-and-forget: update last_used_at (non-blocking per CONTEXT.md)
  admin
    .from('api_keys')
    .update({ last_used_at: new Date().toISOString() })
    .eq('id', data.id)
    .then(() => {})

  return { userId: data.user_id, keyId: data.id }
}
```

Key implementation details:
- Parse Authorization header, reject if not "Bearer vsl_" prefix
- Hash the raw key with SHA-256 to get 64-char hex digest
- Query api_keys table using admin client (bypasses RLS -- per AUTH-04)
- Use timingSafeEqual for defense-in-depth hash comparison (per AUTH-02)
- Fire-and-forget last_used_at update (don't await)

```typescript
import { NextRequest, NextResponse } from 'next/server'
import { authenticateApiKey } from './auth'
import { createAdminClient } from '@/lib/supabase/admin'

export type ApiContext = {
  userId: string
  keyId: string
  supabase: ReturnType<typeof createAdminClient>
}

type ApiHandler = (
  req: NextRequest,
  ctx: ApiContext
) => Promise<NextResponse>

export function withApiAuth(handler: ApiHandler) {
  return async (req: NextRequest) => {
    try {
      const { userId, keyId } = await authenticateApiKey(req)
      const supabase = createAdminClient()
      return await handler(req, { userId, keyId, supabase })
    } catch {
      return NextResponse.json(
        { error: { code: 'UNAUTHORIZED', message: 'Invalid or missing API key' } },
        { status: 401 }
      )
    }
  }
}
```
