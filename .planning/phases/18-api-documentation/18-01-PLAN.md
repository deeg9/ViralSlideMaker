         - POST /carousel — requestBody: CarouselRequest, 200: carousel result, 401/400/402/429
         - GET /templates — 200: { data: Template[] }

         - POST /projects — requestBody: { title, topic?, templateId? }, 201: { data: Project }
         - GET /projects/{id} — 200: { data: Project }
         - DELETE /projects/{id} — 204: no content
         - GET /projects/{id}/slides/{index} — 200: { data: Slide }
         - PATCH /projects/{id}/slides/{index} — requestBody: SlidePatch, 200: { data: Slide }
         - POST /projects/{id}/render — 200: { data: { urls: string[] } }

       - Every path MUST include 401 (unauthorized), 429 (rate limited) error responses
