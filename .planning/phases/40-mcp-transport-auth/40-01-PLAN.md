```typescript
// src/lib/api/auth-core.ts
// Shared SHA-256 + DB lookup + timing-safe compare core for both REST (withApiAuth) and MCP (verifyToken).
// Does NOT depend on NextRequest — operates on a raw bearer string.
import { createHash, timingSafeEqual } from 'crypto'
import { createAdminClient } from '@/lib/supabase/admin'

export class InvalidApiKeyError extends Error {
  constructor(message = 'Invalid API key') {
    super(message)
    this.name = 'InvalidApiKeyError'
  }
}

export async function verifyApiKeyHash(
  rawKey: string
): Promise<{ userId: string; keyId: string }> {
  if (!rawKey.startsWith('vsl_')) {
    throw new InvalidApiKeyError()
  }

  const keyHash = createHash('sha256').update(rawKey).digest('hex')
  const admin = createAdminClient()

  const { data, error } = await admin
    .from('api_keys')
    .select('id, user_id, key_hash')
    .eq('key_hash', keyHash)
    .single()

  if (error || !data) {
    throw new InvalidApiKeyError()
  }

  // Timing-safe compare as defense-in-depth (mirrors original auth.ts behaviour).
  const storedHashBuf = Buffer.from(data.key_hash, 'utf8')
  const computedHashBuf = Buffer.from(keyHash, 'utf8')
  if (storedHashBuf.length !== computedHashBuf.length || !timingSafeEqual(storedHashBuf, computedHashBuf)) {
    throw new InvalidApiKeyError()
  }

  // Fire-and-forget last_used_at update — do NOT await.
  admin
    .from('api_keys')
    .update({ last_used_at: new Date().toISOString() })
    .eq('id', data.id)
    .then(() => {})

  return { userId: data.user_id, keyId: data.id }
}
```

// MCP auth adapter for mcp-handler's withMcpAuth.
// Calls the shared verifyApiKeyHash (NextRequest-free) and the existing Upstash rate limiter.
// Returns AuthInfo on success; undefined on auth failure (→ 401 from withMcpAuth);
// throws TemporarilyUnavailableError on rate-limit failure (→ 503 with temporarily_unavailable errorCode).
import type { AuthInfo } from '@modelcontextprotocol/sdk/server/auth/types.js'
import { TemporarilyUnavailableError } from '@modelcontextprotocol/sdk/server/auth/errors.js'
import { verifyApiKeyHash, InvalidApiKeyError } from '@/lib/api/auth-core'
import { checkRateLimit } from '@/lib/api/rate-limit'

export async function verifyToken(
  _req: Request,
  bearerToken?: string
): Promise<AuthInfo | undefined> {
  if (!bearerToken) return undefined
  if (!bearerToken.startsWith('vsl_')) return undefined

  let userId: string
  let keyId: string
  try {
    ({ userId, keyId } = await verifyApiKeyHash(bearerToken))

  }

  return {
    token: bearerToken,
    clientId: keyId,
    scopes: [],
    extra: { userId, keyId },

}

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-40-01 | Spoofing | withMcpAuth + verifyToken | mitigate | `withMcpAuth(handler, verifyToken, { required: true })` rejects requests with no/malformed Authorization header at the wrapper layer (early return → 401 with WWW-Authenticate header). `verifyToken` returns undefined for missing-bearer / non-vsl-prefix / hash-mismatch / no-row cases — withMcpAuth produces an identical 401 in every case (no info leak). Non-vsl Bearer values never reach the DB. |
| T-40-02 | Denial of Service / Bypass | rate-limit via verifyToken | mitigate | `checkRateLimit(keyId)` is called inside `verifyToken` AFTER hash succeeds — rate-limit is keyed by the verified `keyId` from the api_keys row, NOT by anything user-controllable in headers. Forging the Authorization header cannot bypass the limit because the keyId is server-derived. Same Upstash counter as REST `/api/v1/*` (prefix `vsl:rl`), so cross-protocol abuse is also bounded. On exhaustion, `verifyToken` throws `TemporarilyUnavailableError` → mcp-handler returns MCP-protocol-correct 503 with `temporarily_unavailable` errorCode. |
| T-40-03 | Information Disclosure (timing attack) | verifyApiKeyHash | mitigate | `crypto.timingSafeEqual` used to compare stored vs computed SHA-256 hashes inside `verifyApiKeyHash` (preserved verbatim from the existing src/lib/api/auth.ts implementation; covered by src/__tests__/api/auth-core.test.ts). | T-40-04 | Denial of Service (deploy-time) | mcp-handler module resolution | mitigate | Static top-of-file `import { createMcpHandler, withMcpAuth } from 'mcp-handler'` in `src/app/api/[transport]/route.ts` causes `next build --turbopack` to fail with `Module not found` if mcp-handler is absent from `node_modules`.
