    import type { SupabaseClient } from '@supabase/supabase-js'
    import type { SubscriptionTier } from '@/lib/credits/types'

    // Phase 73 (COST-04): caps are env-driven with locked defaults from CONTEXT.md
    export const FREE_DAILY_CALL_CAP =
      parseInt(process.env.STYLE_EXTRACTION_FREE_DAILY_CALL_CAP ?? '5', 10)
    export const FREE_DAILY_IMAGES_PER_CALL =
      parseInt(process.env.STYLE_EXTRACTION_FREE_DAILY_IMAGES_CAP ?? '3', 10)

    const ROLLING_WINDOW_MS = 24 * 60 * 60 * 1000  // CONTEXT.md: rolling 24h, not calendar day

    export type RateCapReason =
      | 'no_images_provided'
      | 'too_many_images_per_call'
      | 'daily_cap_reached'
      | 'rate_check_failed'

    export interface RateCapResult {
      allowed: boolean
      reason?: RateCapReason
      callsToday: number       // count over the rolling 24h window
      callsRemaining: number   // max(0, cap - callsToday); Infinity for non-free
      imagesPerCallCap: number // FREE_DAILY_IMAGES_PER_CALL or Infinity
    }

    export async function checkExtractionRateCap(
      admin: SupabaseClient,
      userId: string,
      tier: SubscriptionTier,
      requestedImageCount: number,
    ): Promise<RateCapResult> {
      // Premium uncapped per deferred decision — tier !== 'free' shortcuts to allowed
      if (tier !== 'free') {
        return { allowed: true, callsToday: 0, callsRemaining: Infinity, imagesPerCallCap: Infinity }
      }
      if (requestedImageCount <= 0) {
        return { allowed: false, reason: 'no_images_provided', callsToday: 0, callsRemaining: FREE_DAILY_CALL_CAP, imagesPerCallCap: FREE_DAILY_IMAGES_PER_CALL }
      }
      if (requestedImageCount > FREE_DAILY_IMAGES_PER_CALL) {
        return { allowed: false, reason: 'too_many_images_per_call', callsToday: 0, callsRemaining: FREE_DAILY_CALL_CAP, imagesPerCallCap: FREE_DAILY_IMAGES_PER_CALL }
      }
      try {
        const since = new Date(Date.now() - ROLLING_WINDOW_MS).toISOString()
        const { count, error } = await admin
          .from('credit_transactions')
          .select('*', { count: 'exact', head: true })
          .eq('user_id', userId)
          .eq('type', 'gemini_vision_extract_style')
          .gte('created_at', since)
        if (error) {
          console.warn('[rate-cap] DB error; fail-closed:', error)
          return { allowed: false, reason: 'rate_check_failed', callsToday: 0, callsRemaining: 0, imagesPerCallCap: FREE_DAILY_IMAGES_PER_CALL }
        }
        const callsToday = count ?? 0
        const callsRemaining = Math.max(0, FREE_DAILY_CALL_CAP - callsToday)
        if (callsToday >= FREE_DAILY_CALL_CAP) {
          return { allowed: false, reason: 'daily_cap_reached', callsToday, callsRemaining: 0, imagesPerCallCap: FREE_DAILY_IMAGES_PER_CALL }
        }
        return { allowed: true, callsToday, callsRemaining, imagesPerCallCap: FREE_DAILY_IMAGES_PER_CALL }
      } catch (err) {
        console.warn('[rate-cap] threw; fail-closed:', err)
        return { allowed: false, reason: 'rate_check_failed', callsToday: 0, callsRemaining: 0, imagesPerCallCap: FREE_DAILY_IMAGES_PER_CALL }
      }
    }
    ```

    export function isTransientError(err: unknown): boolean {
      if (!err) return false
      const msg = err instanceof Error ? err.message.toLowerCase() : String(err).toLowerCase()
      // Transient signals (heuristic; expand with telemetry):
      if (msg.includes('timeout')) return true
      if (msg.includes('econnreset') || msg.includes('etimedout')) return true
      if (msg.includes('fetch failed') || msg.includes('network')) return true
      // HTTP status from Gemini SDK / fetch — 5xx is transient, 4xx is not
      const status = (err as { status?: number; statusCode?: number })?.status
        ?? (err as { status?: number; statusCode?: number })?.statusCode
      if (typeof status === 'number') return status >= 500 && status < 600
      // Explicit Gemini block reasons are NEVER transient
      if (msg.includes('blocked') || msg.includes('safety') || msg.includes('finish_reason')) return false
      // Zod / validation errors are NEVER transient
      if (msg.includes('zoderror') || msg.includes('invalid_type') || msg.includes('parse')) return false
      return false  // default: do NOT retry on unknown — fail loud per COST-07
    }

    export async function withSingleRetry<T>(fn: () => Promise<T>): Promise<T> {
      try {
        return await fn()
      } catch (err) {
        if (!isTransientError(err)) throw err
        console.warn('[extract-retry] transient error; ONE retry attempt:', err)
        return await fn()  // SECOND attempt — that's it. No third try.
      }
    }
    ```

      - isTransientError: 400 status → false
      - isTransientError: SAFETY block → false
      - isTransientError: ZodError → false
      - isTransientError: unknown error → false (fail-loud default)
      - withSingleRetry: fn succeeds first try → fn called once, returns value

      - withSingleRetry: fn throws permanent → fn called once, throws immediately (NO retry)

    function verifyCronSecret(req: Request): boolean {

      const expected = `Bearer ${process.env.CRON_SECRET ?? ''}`
      return process.env.CRON_SECRET ? auth === expected : false
    }

    async function runBillAlarm() {
      const admin = createAdminClient()
      const startOfDayUtc = new Date()
      startOfDayUtc.setUTCHours(0, 0, 0, 0)
      const since = startOfDayUtc.toISOString()

      // Sum NET extraction calls today (charges - refunds where description references extract-style)
      const { count: chargesToday, error: chargeErr } = await admin
        .from('credit_transactions')
        .select('*', { count: 'exact', head: true })
        .eq('type', 'gemini_vision_extract_style')
        .gte('created_at', since)

      const { count: refundsToday, error: refundErr } = await admin
        .from('credit_transactions')
        .select('*', { count: 'exact', head: true })
        .eq('type', 'refund')
        .like('description', 'Refund: extract-style%')
        .gte('created_at', since)

      if (chargeErr || refundErr) {
        console.error('[bill-alarm] DB error; cannot compute today cost:', { chargeErr, refundErr })
        return { ok: false, error: 'db_error' }
      }

      const netCalls = Math.max(0, (chargesToday ?? 0) - (refundsToday ?? 0))
      const todayCostUsd = netCalls * PER_CALL_COST_USD
      const breach = todayCostUsd > DAILY_COST_CEILING_USD

      const payload = {
        date: since.slice(0, 10),
        netCalls,
        chargesToday: chargesToday ?? 0,
        refundsToday: refundsToday ?? 0,
        todayCostUsd: Number(todayCostUsd.toFixed(4)),
        ceilingUsd: DAILY_COST_CEILING_USD,
        breach,
      }

      if (breach) {
        // Pre-PMF: structured warn is sufficient (Vercel monitoring captures it).
        // Future: add Slack/email delivery here — interface stays the same.
        console.warn('[bill-alarm] CEILING BREACHED', JSON.stringify(payload))
      } else {
        console.log('[bill-alarm] within ceiling', JSON.stringify(payload))
      }
      return { ok: true, ...payload }
    }

    export async function GET(req: Request) {
      if (!verifyCronSecret(req)) {
        return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
      }
      const result = await runBillAlarm()
      return NextResponse.json(result, { status: result.ok ? 200 : 500 })
    }

    export async function POST(req: Request) {
      // Manual invocation in dev; same auth gate
      if (!verifyCronSecret(req)) {
        return NextResponse.json({ error: 'unauthorized' }, { status: 401 })
      }
      const result = await runBillAlarm()
      return NextResponse.json(result, { status: result.ok ? 200 : 500 })
    }
