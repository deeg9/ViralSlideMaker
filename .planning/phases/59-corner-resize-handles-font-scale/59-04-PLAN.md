### react-rnd onResizeStop callback signature (already used in 59-02)

```typescript
onResizeStop?: (
  e: MouseEvent | TouchEvent,    // ← read e.shiftKey here
  dir: ResizableDirection,
  ref: HTMLElement,
  delta: { width: number; height: number },
  position: { x: number; y: number },
) => void
```

`MouseEvent.shiftKey` is `boolean` (true while Shift is held during the gesture's pointerup). `TouchEvent` does NOT have `.shiftKey` (touch devices have no Shift modifier — the property is undefined and falsy, which is exactly the desired behavior on touch).

### Existing C2 onResizeStop dispatch shape (slide-preview.tsx, after plan 59-02)

```tsx
onResizeStop={(_e, dir, ref, _delta, position) => {
  const newWidthPx = parseInt(ref.style.width)
  const newWidthPct = Math.max(20, Math.min(100, (newWidthPx / REF_WIDTH) * 100))
  const isCorner = isCornerDir(dir as RndDir)
  let newFontSize: number | undefined = undefined
  if (isCorner && hookResizeStartRef.current) {
    newFontSize = scaleFontSizeOnCorner(...)
  }
  // ...
}}
```

C4 changes the `if (isCorner && ...)` guard to `if (isCorner && !e.shiftKey && ...)`. The `_e` param underscore is removed to indicate the param is now read.

```tsx
onResizeStop={(e, dir, ref, _delta, position) => {
  const newWidthPx = parseInt(ref.style.width)
  const newWidthPct = Math.max(20, Math.min(100, (newWidthPx / REF_WIDTH) * 100))
  const isCorner = isCornerDir(dir as RndDir)
  let newFontSize: number | undefined = undefined
  // phase-59-c4: Shift+corner = width-only escape (RESIZE-04). When Shift is
  // held during a corner gesture, skip the font_size dispatch and behave like
  // a side handle. TouchEvent has no shiftKey so this is no-op on touch.
  if (isCorner && !(e as MouseEvent).shiftKey && hookResizeStartRef.current) {
    newFontSize = scaleFontSizeOnCorner(
      hookResizeStartRef.current.fontSize,
      hookResizeStartRef.current.widthPx,
      newWidthPx,
      'hook',
    )
  }
  if (onSizeChange) {
    onSizeChange('hook', {
      width: Math.round(newWidthPct),
      ...(newFontSize !== undefined ? { font_size: newFontSize } : {}),
    })
  } else {
    onWidthChange?.('hook', Math.round(newWidthPct))
  }
  if (onPositionChange) {
    const percentX = Math.max(0, Math.min(100, (position.x / REF_WIDTH) * 100))
    const percentY = Math.max(0, Math.min(100, (position.y / REF_HEIGHT) * 100))
    onPositionChange('hook', { x: percentX, y: percentY })
  }
  hookResizeStartRef.current = null
}}
```

  // simulates the gesture by firing pointer events with shiftKey on/off.

  // The primary contract: when shiftKey is true and dir is a corner, the

  // false, it includes font_size. Side handles never dispatch font_size

  function makeFixture() {
    const slide: Slide = {
      hook_text: 'STOP SCROLLING',
      body_text: 'Body',
      image_query: 'q',

      background_url: null,
      pexels_photo_id: null,

    return { slide, template: getTemplate('highlight-bar') }
  }

  it('Shift+TL corner: dispatched payload contains width, no font_size', () => {
    const { slide, template } = makeFixture()
    const onSizeChange = vi.fn()

    const { container } = render(
      <SlidePreview
        slide={slide}
        template={template}
        index={0}
        editable
        onSizeChange={onSizeChange}
      />,
    )

    // Find the topLeft corner handle (react-rnd applies our resizeHandleStyles
    // to its internal handle <span>s; the cursor:'nwse-resize' inline style is

    const handles = Array.from(container.querySelectorAll('span, div')).filter(
      (el) => (el as HTMLElement).style?.cursor === 'nwse-resize',

    // Expect at least 2 (TL + BR per text box; could be more across multiple boxes)
