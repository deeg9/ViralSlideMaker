    /**
     * Phase 60 (SEL-13): Pure union-AABB computation for the multi-select ring.
     *
     * No React or DOM. Reusable from any layer (server, RPC, future MCP tool).
     * Mirrors text-box-adapter.ts pure-module convention.
     *
     * Phase 60 known gaps (TODO when later phases revisit):
     *  - Structural-anchor TextBoxes (anchor !== 'custom') are SKIPPED — height
     *    resolution requires DOM ref via the rendered-height observer pattern at
     *    slide-preview.tsx:589-637. Phase 63 (alignment) will lift that into a
     *    shared helper.
     *  - Watermarks are SKIPPED — size is computed by resolveWatermark /
     *    resolveForcedWatermark at render time. Phase 62/63 revisit.
     */

    import type { Slide } from '@/lib/types/slide'

    export interface AABB {
      x: number       // % of canvas width
      y: number       // % of canvas height
      width: number   // % of canvas width
      height: number  // % of canvas height
    }

    export function computeSelectionBounds(slide: Slide, ids: string[]): AABB | null {
      if (ids.length === 0) return null

      const idSet = new Set(ids)
      const rects: AABB[] = []

      for (const tb of slide.text_boxes ?? []) {
        if (!idSet.has(tb.id)) continue
        if (tb.position.anchor !== 'custom') continue  // Phase 60 known gap (see header)
        rects.push({
          x: tb.position.x ?? 0,
          y: tb.position.y ?? 0,
          width: tb.position.width ?? 60,
          height: 0,  // Phase 60 known gap — DOM-ref-based resolution lands later
        })
      }

      for (const ci of slide.custom_images ?? []) {
        if (!idSet.has(ci.id)) continue
        rects.push({ x: ci.x, y: ci.y, width: ci.width, height: ci.height })
      }

      // Watermark intentionally skipped — see header comment.

      if (rects.length === 0) return null

      const minX = Math.min(...rects.map(r => r.x))
      const minY = Math.min(...rects.map(r => r.y))
      const maxX = Math.max(...rects.map(r => r.x + r.width))
      const maxY = Math.max(...rects.map(r => r.y + r.height))
      return { x: minX, y: minY, width: maxX - minX, height: maxY - minY }
    }
    ```

    import type { Slide } from '@/lib/types/slide'
    import type { Selection } from '@/lib/editor/use-selection-model'

    export interface MultiSelectTypeBarProps {
      slide: Slide
      selection: Selection
      onBulkDelete: (ids: string[]) => void
      // Phase 60 only declares this prop set. Bulk-edit handlers
      // (font, color, treatment) wire in Phase 63 (alignment).
    }

    /**
     * Phase 60 (SEL-12): Right-rail panel shown when N >= 2 elements are
     * selected. Replaces the single-element ActiveTextBoxCard via a ternary
     * branch in text-editor-panel.tsx.
     *
     * Phase 60 minimal surface: count + type breakdown + Delete button.
     * Property fields (font, color, treatment) arrive in Phase 63 alongside
     * the alignment toolbar.
     *
     * NOTE per PITFALL-12.3 / PITFALL-1.3: no "primary" key-object UX.
     * Right rail does NOT track which element was clicked first.
     */
    export function MultiSelectTypeBar({ slide, selection, onBulkDelete }: MultiSelectTypeBarProps) {
      const idSet = new Set(selection.ids)
      const textBoxCount = (slide.text_boxes ?? []).filter(tb => idSet.has(tb.id)).length
      const imageCount = (slide.custom_images ?? []).filter(ci => idSet.has(ci.id)).length
      const watermarkCount = slide.watermark && idSet.has(slide.watermark.id) ? 1 : 0
      const total = selection.ids.length

      return (
        <div className="space-y-3 rounded-md border border-neutral-200 bg-white p-3">
          <h3 className="text-sm font-medium">{total} elements selected</h3>
          <ul className="space-y-1 text-xs text-neutral-700">
            {textBoxCount > 0 && <li>{textBoxCount} {textBoxCount === 1 ? 'text box' : 'text boxes'}</li>}
            {imageCount > 0 && <li>{imageCount} {imageCount === 1 ? 'image' : 'images'}</li>}
            {watermarkCount > 0 && <li>1 watermark</li>}
          </ul>
          <button
            type="button"
            onClick={() => onBulkDelete(selection.ids)}
            className="w-full rounded-md border border-red-300 bg-red-50 px-2 py-1 text-xs text-red-700 hover:bg-red-100"
          >
            Delete {total} elements
          </button>
        </div>
      )
    }
    ```
