```typescript
// src/__tests__/render/new-templates-parity.test.tsx
import { describe, it, expect } from 'vitest'
import { renderToString } from 'react-dom/server'
import { templates, getTemplate } from '@/lib/render/templates'
import { SlidePreview } from '@/components/slides/slide-preview'  // verify import path
import { buildSlideElement } from '@/lib/render/render-slide-satori'
import { makeSampleSlide } from '@/lib/render/sample-slide'

const NEW_TEMPLATE_IDS = ['editorial-serif', 'notes-app-chic', 'brutalist-grid'] as const

describe('Phase 56: new templates parity (TPL-10)', () => {
  for (const id of NEW_TEMPLATE_IDS) {
    describe(`${id}`, () => {
      it('is registered in templates array', () => {
        const tpl = getTemplate(id)
        expect(tpl).toBeDefined()
        expect(tpl?.id).toBe(id)
      })

      it('React preview HTML and Satori-target output share identical resolver helper signatures', () => {
        const tpl = getTemplate(id)!
        const slide = makeSampleSlide(tpl)
        // Render React preview to HTML string
        const reactHTML = renderToString(<SlidePreview slide={slide} template={tpl} slideIndex={0} />)
        // Render Satori-shaped element (we don't actually run resvg — just assert the JSX tree)
        const satoriElement = buildSlideElement(slide, tpl, null, null, undefined, { slideIndex: 0, totalSlides: 1, topic: undefined })
        // Both renderers should reference the same data-layer markers when grain/vignette/decorative are declared
        if (tpl.grainMode) {
          expect(reactHTML).toContain('data-layer="grain"')
          // satoriElement is a React element — convert to string via renderToString
          const satoriHTML = renderToString(satoriElement as React.ReactElement)
          expect(satoriHTML).toContain('data-layer="grain"')
        }
        if (tpl.vignette) {
          expect(reactHTML).toContain('data-layer="vignette"')
        }
        // Decorative overlays from template.overlays[] should appear in BOTH renderers
        const decorativeOverlays = (tpl.overlays ?? []).filter(o => o.content.type === 'decorative')
        for (const ov of decorativeOverlays) {
          if (ov.content.type === 'decorative') {
            expect(reactHTML).toContain(`data-asset="${ov.content.asset}"`)
            const satoriHTML = renderToString(satoriElement as React.ReactElement)
            expect(satoriHTML).toContain(`data-asset="${ov.content.asset}"`)
          }
        }
      })

      it('configuration is byte-identical to a committed snapshot', () => {
        const tpl = getTemplate(id)!
        const json = JSON.stringify(tpl, Object.keys(tpl).sort(), 2)
        // Snapshot fixture committed; first run creates it, subsequent runs compare
        expect(json).toMatchSnapshot()
      })
    })
  }
})
```

```tsx
const NEW_TEMPLATE_IDS = ['editorial-serif', 'notes-app-chic', 'brutalist-grid'] as const

describe('Phase 56: new templates parity (TPL-10)', () => {

      it('is registered in the templates array', () => {
        const tpl = getTemplate(id)
        expect(tpl).toBeDefined()
        expect(tpl?.id).toBe(id)
      })

      it('declares an explicit bodyFontFamily (D-04 free-tier discipline)', () => {
        const tpl = getTemplate(id)!
        expect(tpl.bodyFontFamily).toBeDefined()
        expect(tpl.bodyFontFamily).not.toBe('')
        // Must NOT be Inter (D-39, Anti-Pattern #1)
        expect(tpl.bodyFontFamily).not.toContain('Inter')
      })

      it('declares coverLayout distinct from default layout (TPL-06)', () => {
        const tpl = getTemplate(id)!
        expect(tpl.coverLayout).toBeDefined()
      })

      it('is NOT in PREMIUM_TEMPLATE_IDS (free-tier per TPL-09)', async () => {
        const { PREMIUM_TEMPLATE_IDS } = await import('@/lib/render/templates/premium-tier')
        expect((PREMIUM_TEMPLATE_IDS as readonly string[]).includes(id)).toBe(false)
      })

      it('thumbnail path matches /templates/{id}.png convention (TPL-08)', () => {
        const tpl = getTemplate(id)!
        expect(tpl.thumbnail).toBe(`/templates/${id}.png`)
      })

      it('React preview HTML includes data-layer markers for all declared visual layers (TPLREND-05)', () => {
        const tpl = getTemplate(id)!
        const slide = makeSampleSlide(tpl)
        const html = renderToString(<SlidePreview slide={slide} template={tpl} slideIndex={0} />)
        if (tpl.grainMode) {
          expect(html).toContain('data-layer="grain"')
        }
        if (tpl.vignette) {
          expect(html).toContain('data-layer="vignette"')
        }
      })

      it('Satori-target JSX tree includes data-layer markers matching React preview (TPLREND-05 parity)', () => {
        const tpl = getTemplate(id)!
        const slide = makeSampleSlide(tpl)
        const ctx = { slideIndex: 0, totalSlides: 1, topic: undefined as string | undefined }
        const satoriElement = buildSlideElement(slide, tpl, null, null, undefined, ctx)
        const satoriHTML = renderToString(satoriElement as React.ReactElement)
        if (tpl.grainMode) {
          expect(satoriHTML).toContain('data-layer="grain"')
        }
        if (tpl.vignette) {
          expect(satoriHTML).toContain('data-layer="vignette"')
        }
      })

      it('decorative overlays appear in BOTH renderers when declared (TPL-04)', () => {
        const tpl = getTemplate(id)!

        if (decorativeOverlays.length === 0) return  // brutalist-grid has none — skip
        const slide = makeSampleSlide(tpl)
        const reactHTML = renderToString(<SlidePreview slide={slide} template={tpl} slideIndex={0} />)
        const ctx = { slideIndex: 0, totalSlides: 1, topic: undefined as string | undefined }

        for (const ov of decorativeOverlays) {
          if (ov.content.type === 'decorative') {

      })
    })
  }
```
