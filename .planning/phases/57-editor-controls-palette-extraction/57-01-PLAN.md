### T2 — Create palette.ts

File: `src/lib/photo-treatment/palette.ts`

```ts
// SERVER-ONLY: do not import from client components.
// Phase 57 (PALETTE-01, PALETTE-04): named-role palette extraction
// for palette-dominant templates (editorial-serif, notes-app-chic).

import Vibrant from 'node-vibrant/node'
import { sharp } from './sharp-config'

export type NamedPalette = {
  Vibrant: string
  DarkVibrant: string
  LightVibrant: string
  Muted: string
  DarkMuted: string
  LightMuted: string
}

const FALLBACK_HEX = '#7d6161' // brand muted (final cascade fallback)

const ROLE_CASCADE: Array<keyof NamedPalette> = [
  'Vibrant',
  'DarkVibrant',
  'Muted',
  'DarkMuted',
  'LightVibrant',
  'LightMuted',
]

export async function extractPalette(treatedUrl: string): Promise<NamedPalette> {
  // 1. Fetch the treated URL as a buffer
  const response = await fetch(treatedUrl)
  if (!response.ok) {
    throw new Error(`extractPalette: fetch failed ${response.status} for ${treatedUrl}`)
  }
  const arrayBuffer = await response.arrayBuffer()
  const sourceBuffer = Buffer.from(arrayBuffer)

  // 2. Resize via sharp to ≤256×256 (PALETTE-04 budget)
  const thumbBuffer = await sharp(sourceBuffer)
    .resize(256, 256, { fit: 'inside', withoutEnlargement: true })
    .png()
    .toBuffer()

  // 3. Pass to Vibrant
  const palette = await Vibrant.from(thumbBuffer).getPalette()

  // 4. Map roles with cascade fallback
  const raw: Partial<NamedPalette> = {
    Vibrant: palette.Vibrant?.hex,
    DarkVibrant: palette.DarkVibrant?.hex,
    LightVibrant: palette.LightVibrant?.hex,
    Muted: palette.Muted?.hex,
    DarkMuted: palette.DarkMuted?.hex,
    LightMuted: palette.LightMuted?.hex,
  }

  function resolveRole(key: keyof NamedPalette): string {
    if (raw[key]) return raw[key]!
    for (const fallback of ROLE_CASCADE) {
      if (raw[fallback]) return raw[fallback]!
    }
    return FALLBACK_HEX
  }

  return {
    Vibrant: resolveRole('Vibrant'),
    DarkVibrant: resolveRole('DarkVibrant'),
    LightVibrant: resolveRole('LightVibrant'),
    Muted: resolveRole('Muted'),
    DarkMuted: resolveRole('DarkMuted'),
    LightMuted: resolveRole('LightMuted'),
  }
}
```

### T5 — Add 5 resolver helpers to resolve-overlay.ts

File: `src/lib/render/resolve-overlay.ts`

Append these 5 pure helpers (do NOT touch existing exports):

```ts
import type { Slide } from '@/lib/types/slide'
import type { SlideTemplate } from '@/lib/render/templates/index'

// Phase 57 (D-07, D-16): palette-aware rendering helpers.
// Both render-slide-satori.tsx and slide-preview.tsx import from this
// module to honor TPLREND-05 (no JSX duplicated across renderers).

export function getVibrant(slide: Slide): string | null {
  return slide.style_overrides?.derived_palette?.[0] ?? null
}

export function getDarkMuted(slide: Slide): string | null {
  return slide.style_overrides?.derived_palette?.[4] ?? null
}

export function getEditorialSerifChipColors(
  slide: Slide,
  template: SlideTemplate,
): { background: string; text: string } {
  if (template.paletteMode !== 'palette-dominant') {
    return { background: '#F5EFE6', text: '#3d0000' }
  }
  const vibrant = getVibrant(slide)
  if (!vibrant) {
    return { background: '#F5EFE6', text: '#3d0000' }
  }
  const darkMuted = getDarkMuted(slide) ?? '#3d0000'
  return { background: vibrant, text: darkMuted }
}

export function getNotesAppChicAccentColor(
  slide: Slide,
  template: SlideTemplate,
): string {
  if (template.paletteMode !== 'palette-dominant') return '#FFE066'
  return getVibrant(slide) ?? '#FFE066'
}

export function resolveGrainIntensity(
  slide: Slide,
  template: SlideTemplate,
): 'subtle' | 'medium' | 'heavy' | 'off' | undefined {
  // D-09: per-slide override on top of template.grainMode (Phase 56).
  // 'off' explicitly disables grain. Undefined falls back to template.
  const override = slide.style_overrides?.grain_intensity
  if (override) return override
  // Map template.grainMode (Phase 55 enum: 'subtle'|'medium'|'heavy'|'paper'|'xerox')
  // — only the 4 simple values are user-overridable; paper/xerox stay template-only.
  if (template.grainMode === 'subtle' || template.grainMode === 'medium' || template.grainMode === 'heavy') {
    return template.grainMode
  }
  return undefined
}
```
