    `src/components/style-from-reference/style-from-reference-shell.tsx`:
    ```tsx
    'use client'
    import { useReducer, useEffect, useCallback } from 'react'
    import { ReferenceImageUploader } from '@/components/dashboard/reference-image-uploader'
    import { extractStyleSpecAction } from '@/app/(dashboard)/style-from-reference/actions'
    import { specPreviewReducer, initialSpecPreviewState } from './state-machine'
    import { StickyReferenceStrip } from './sticky-reference-strip'
    import { SpecPreviewPanel } from './spec-preview-panel'
    import type { BrandKit } from '@/lib/types/brand-kit'

    export interface StyleFromReferenceShellProps {
      projectId: string
      brandKit: BrandKit | null
      brandKitAvailable: boolean
    }

    export function StyleFromReferenceShell(props: StyleFromReferenceShellProps) {
      const [state, dispatch] = useReducer(specPreviewReducer, initialSpecPreviewState)
      // Hydrate brand-kit availability into reducer on mount
      useEffect(() => { dispatch({ type: 'BRAND_KIT_AVAILABILITY', available: props.brandKitAvailable }) }, [props.brandKitAvailable])

      const handleExtract = useCallback(async (rows) => {
        dispatch({ type: 'EXTRACT_START' })
        const result = await extractStyleSpecAction({ projectId: props.projectId, references: rows })
        if (result.ok) dispatch({ type: 'EXTRACT_SUCCESS', spec: result.spec, refusalFlags: result.refusal_flags, telemetry: result.telemetry })
        else dispatch({ type: 'EXTRACT_ERROR', error: { kind: result.error, detail: 'detail' in result ? String(result.detail) : undefined, recovery_suggestion: 'recovery_suggestion' in result ? result.recovery_suggestion : undefined }, telemetry: result.telemetry })
      }, [props.projectId])

      const handleReExtract = useCallback(async () => {
        dispatch({ type: 'RE_EXTRACT_CLICK' })
        // After RE_EXTRACT_CLICK the reducer is in 'extracting' (if retries remained).
        // Re-call action with the SAME state.references.
        const result = await extractStyleSpecAction({ projectId: props.projectId, references: state.references })
        if (result.ok) dispatch({ type: 'EXTRACT_SUCCESS', ... })
        else dispatch({ type: 'EXTRACT_ERROR', ... })
      }, [state.references, props.projectId])

      return (
        <div className="space-y-6">
          {/* PREVIEW-09: sticky strip rendered OUTSIDE conditional blocks — visible across all states */}
          <StickyReferenceStrip references={state.references} />

          {state.status === 'idle' && (
            <ReferenceImageUploader
              onUploadComplete={(row) => dispatch({ type: 'UPLOAD_ADDED', row })}
              onExtractRequested={(rows) => { void handleExtract(rows) }}
            />
          )}

          {state.status === 'extracting' && (
            <div data-testid="extracting-state" role="status" aria-live="polite">
              Analyzing your references...
            </div>
          )}

          {state.status === 'preview' && (
            <SpecPreviewPanel
              state={state}
              dispatch={dispatch}
              brandKit={props.brandKit}
              onGenerate={() => dispatch({ type: 'GENERATE_CLICK' })}
              onReExtract={() => { void handleReExtract() }}
            />
          )}

          {state.status === 'generating' && (
            <div data-testid="generating-state" role="status">Generating your slideshow...</div>
          )}

          {state.error && (
            <div data-testid="error-mount">
              {/* Plan 76-03 fills this with RefusalWarning + SafetyBlockRecovery */}
              {state.error.kind}
            </div>
          )}
        </div>
      )
    }
    ```
