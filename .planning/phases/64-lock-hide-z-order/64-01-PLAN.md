            idSet.has(tb.id) && (tb.locked ?? false) !== action.locked
              ? { ...tb, locked: action.locked }
              : tb
          )
          if (candidate.some((tb, j) => tb !== slide.text_boxes![j])) {
            nextTextBoxes = candidate
            changed = true
          }
        }
        let nextCustomImages = slide.custom_images
        if (slide.custom_images) {
          const candidate = slide.custom_images.map(ci =>
            idSet.has(ci.id) && (ci.locked ?? false) !== action.locked
              ? { ...ci, locked: action.locked }
              : ci
          )
          if (candidate.some((ci, j) => ci !== slide.custom_images![j])) {
            nextCustomImages = candidate
            changed = true
          }
        }
        let nextWatermark = slide.watermark
        if (slide.watermark && idSet.has(slide.watermark.id) &&
            (slide.watermark.locked ?? false) !== action.locked) {
          nextWatermark = { ...slide.watermark, locked: action.locked }
          changed = true
        }
        if (nextTextBoxes === slide.text_boxes &&
            nextCustomImages === slide.custom_images &&
            nextWatermark === slide.watermark) return slide
