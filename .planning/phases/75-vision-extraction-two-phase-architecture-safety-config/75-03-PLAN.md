    **`extractPaletteFromBytes(bytes: Buffer): Promise<ExtractedPalette>`:**
    ```typescript
    import { Vibrant } from 'node-vibrant/node'
    // Note: subpath import is REQUIRED — bare 'node-vibrant' resolves to a throw-on-import stub
    // per package.json exports field (./dist/esm/throw.js). Use 'node-vibrant/node' on server.

    export async function extractPaletteFromBytes(bytes: Buffer): Promise<ExtractedPalette> {
      try {
        const palette = await Vibrant.from(bytes).getPalette()
        // Order: try Vibrant > LightVibrant > Muted > DarkVibrant > DarkMuted > LightMuted (perceptual saliency order — Vibrant docs)
        const swatchOrder = ['Vibrant', 'LightVibrant', 'Muted', 'DarkVibrant', 'DarkMuted', 'LightMuted'] as const
        const swatches: Array<{ hex: string; population: number }> = []
        for (const name of swatchOrder) {
          const sw = palette[name]
          if (sw) {
            swatches.push({ hex: normalizeHex(sw.hex), population: sw.population ?? 0 })
          }
        }

        if (swatches.length === 0) {
          return { hex: [FALLBACK_NEUTRAL_HEX], dominant: FALLBACK_NEUTRAL_HEX }
        }

        // Dedupe (Vibrant can return the same hex for multiple roles on flat images)
        const uniqueHex: string[] = []
        const seen = new Set<string>()
        for (const s of swatches) {
          if (!seen.has(s.hex)) { uniqueHex.push(s.hex); seen.add(s.hex) }
        }

        // Dominant = swatch with highest population (the most pixels)
        const dominant = swatches.reduce((best, cur) => cur.population > best.population ? cur : best, swatches[0]).hex

        return { hex: uniqueHex, dominant }
      } catch (err) {
        console.warn('[palette-extraction] Vibrant.from threw (graceful fallback):', err)
        return { hex: [FALLBACK_NEUTRAL_HEX], dominant: FALLBACK_NEUTRAL_HEX }
      }
    }

    function normalizeHex(input: string): string {
      const raw = input.startsWith('#') ? input.slice(1) : input
      const expanded = raw.length === 3
        ? raw.split('').map(c => c + c).join('')
        : raw.padEnd(6, '0').slice(0, 6)
      return ('#' + expanded).toUpperCase()
    }
    ```

    ```typescript
    export function mergePalettes(palettes: ExtractedPalette[]): ExtractedPalette {
      if (palettes.length === 0) {
        return { hex: [FALLBACK_NEUTRAL_HEX], dominant: FALLBACK_NEUTRAL_HEX }
      }

      // Step 1: count dominant votes — winner is the dominant of the merged palette.
      // Tiebreaker: lexicographically-lower normalized hex wins (deterministic per Test 7).
      const dominantVotes = new Map<string, number>()
      for (const p of palettes) {
        const norm = normalizeHex(p.dominant)
        dominantVotes.set(norm, (dominantVotes.get(norm) ?? 0) + 1)
      }
      const sortedByVotes = [...dominantVotes.entries()].sort(([hexA, votesA], [hexB, votesB]) => {
        if (votesB !== votesA) return votesB - votesA
        return hexA.localeCompare(hexB)
      })
      const mergedDominant = sortedByVotes[0][0]

      // Step 2: count per-hex appearances; sort by frequency desc, mergedDominant first, then lex asc. Cap at PALETTE_TARGET_SIZE.
      const hexCounts = new Map<string, number>()
      for (const p of palettes) {
        for (const h of p.hex) {
          const norm = normalizeHex(h)
          hexCounts.set(norm, (hexCounts.get(norm) ?? 0) + 1)
        }
      }
      const sortedHexes = [...hexCounts.entries()].sort(([hexA, ca], [hexB, cb]) => {
        if (hexA === mergedDominant && hexB !== mergedDominant) return -1
        if (hexB === mergedDominant && hexA !== mergedDominant) return 1
        if (cb !== ca) return cb - ca
        return hexA.localeCompare(hexB)
      })
      const mergedHex = sortedHexes.slice(0, PALETTE_TARGET_SIZE).map(([hex]) => hex)

      return { hex: mergedHex, dominant: mergedDominant }
    }
    ```
